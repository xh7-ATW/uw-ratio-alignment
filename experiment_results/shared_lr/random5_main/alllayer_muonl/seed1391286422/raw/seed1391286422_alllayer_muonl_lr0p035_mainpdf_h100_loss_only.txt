#!/usr/bin/env python3
"""
train_alignment.py

Draft rerun script for the Muon / MuonL relative-step schemes:
  plain_muon    : x_i = r_i
  typed_muonl   : x_i = rbar_C, the same-type mean of r_i
  alllayer_muonl: x_i = rbar_all, the all-matrix mean of r_i

Seed defaults to 42 and can be overridden with MYOPT_SEED. The training implementation is copied from the previous
uw_ratio_alignment implementation, with this rerun-specific wrapper:
  - only plain_muon, typed_muonl, and alllayer_muonl are accepted
  - default matrix lr is 0.03 and matrix weight decay is 0.025
  - the linear cooldown window defaults to the final 25% of schedule steps

When MYOPT_DIAG_MODE=local_quadratic, every DIAG_FREQ training steps it records the
Theory-PDF quantities around U-W ratio alignment:
  - r_i = ||U_i||_F / ||W_i||_F and candidate x_i values
  - <G_i,V_i>, <V_i,H_ii V_i>, and optional C_ij=<V_i,H_ij V_j>
  - enough fields for score and sufficient-criterion postprocessing

When MYOPT_DIAG_MODE=orscale, it keeps the previous OrScale diagnostic:
  - per-block alpha_i, a_i=||G_i||_*, L_i, b_i=L_i d_i, y_i, z_i
  - checkpoint Phi, kappa ratios, cos2, nu, and rho for plain/typed/all
  - cached/warm-started L_i estimates so curvature is not recomputed from
    scratch at every diagnostic checkpoint
  - by default uses a microbatch empirical-Fisher Gram L_i proxy; exact
    block-GGN power iteration remains available but is much slower
  - no q/E/S/D/B stationarity diagnostics

Control via environment variable:
  MYOPT_SCHEME = plain_muon | typed_muonl | alllayer_muonl

Output files (in this directory):
  seed{MYOPT_SEED}_{SCHEME}_orscale.txt       training log
  seed{MYOPT_SEED}_{SCHEME}_orscale_diag.json diagnostic records (list, one entry per checkpoint)
"""

def _load_data_shard(file: Path):
    header = torch.from_file(str(file), False, 256, dtype=torch.int32)
    assert header[0] == 20240520
    assert header[1] == 1
    num_tokens = int(header[2])
    with file.open("rb", buffering=0) as f:
        tokens = torch.empty(num_tokens, dtype=torch.uint16, pin_memory=True)
        f.seek(256 * 4)
        nbytes = f.readinto(tokens.numpy())
        assert nbytes == 2 * num_tokens
    return tokens

def distributed_data_generator(filename_pattern: str, batch_size: int, seq_len=1024):
    files = sorted(Path(p) for p in glob.glob(str(filename_pattern)))
    if not files:
        files = sorted(Path(p) for p in glob.glob(str(REPO_ROOT / filename_pattern)))
    assert files, f"no data shards match {filename_pattern!r}"
    assert batch_size % dist.get_world_size() == 0
    local_batch_size = batch_size // dist.get_world_size()
    file_iter = iter(files)
    tokens, pos = _load_data_shard(next(file_iter)), 0
    while True:
        if pos + batch_size + 1 >= len(tokens):
            tokens, pos = _load_data_shard(next(file_iter)), 0
        buf = tokens[pos + dist.get_rank() * local_batch_size:][:local_batch_size + 1]
        inputs = buf[:-1].to(device="cuda", dtype=torch.int32, non_blocking=True)
        targets = buf[1:].to(device="cuda", dtype=torch.int64, non_blocking=True)
        pos += batch_size
        yield inputs.view(-1, seq_len), targets.view(-1, seq_len)

class RMSNorm(nn.Module):
    def __init__(self, dim):
        super().__init__()
        self.gains = nn.Parameter(torch.ones(dim))

    def forward(self, x):
        return F.rms_norm(x, (x.size(-1),), weight=self.gains.type_as(x))

class Linear(nn.Linear):
    def __init__(self, in_features, out_features):
        super().__init__(in_features, out_features, bias=True)

    def forward(self, x):
        return F.linear(x, self.weight.type_as(x), self.bias.type_as(x))

class Rotary(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        angular_freq = (1 / 1024) ** torch.linspace(0, 1, steps=dim//4, dtype=torch.float32)
        self.register_buffer("angular_freq", torch.cat([angular_freq, angular_freq.new_zeros(dim//4)]))

    def forward(self, x_BTHD: Tensor):
        pos = torch.arange(x_BTHD.size(1), dtype=torch.float32, device=x_BTHD.device)
        theta = torch.outer(pos, self.angular_freq)[None, :, None, :]
        cos, sin = theta.cos(), theta.sin()
        x1, x2 = x_BTHD.to(dtype=torch.float32).chunk(2, dim=-1)
        y1 = x1 * cos + x2 * sin
        y2 = x1 * (-sin) + x2 * cos
        return torch.cat((y1, y2), 3).type_as(x_BTHD)

class CausalSelfAttention(nn.Module):
    def __init__(self, dim: int, head_dim=128):
        super().__init__()
        self.num_heads = dim // head_dim
        self.head_dim = head_dim
        hdim = self.num_heads * self.head_dim
        self.q = Linear(dim, hdim)
        self.k = Linear(dim, hdim)
        self.v = Linear(dim, hdim)
        self.proj = Linear(hdim, dim)
        self.rotary = Rotary(head_dim)

    def forward(self, x: Tensor):
        B, T = x.size(0), x.size(1)
        q = self.q(x).view(B, T, self.num_heads, self.head_dim)
        k = self.k(x).view(B, T, self.num_heads, self.head_dim)
        v = self.v(x).view(B, T, self.num_heads, self.head_dim)
        q, k = F.rms_norm(q, (q.size(-1),)), F.rms_norm(k, (k.size(-1),))
        q, k = self.rotary(q), self.rotary(k)
        y = F.scaled_dot_product_attention(q.transpose(1, 2), k.transpose(1, 2),
                                           v.transpose(1, 2), scale=0.12, is_causal=True).transpose(1, 2)
        y = y.contiguous().view(B, T, self.num_heads * self.head_dim)
        return self.proj(y)

class MLP(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        hdim = 4 * dim
        self.fc = Linear(dim, hdim)
        self.proj = Linear(hdim, dim)

    def forward(self, x: Tensor):
        return self.proj(self.fc(x).relu().square())

class Block(nn.Module):
    def __init__(self, dim: int):
        super().__init__()
        self.attn = CausalSelfAttention(dim)
        self.mlp = MLP(dim)
        self.norm1 = RMSNorm(dim)
        self.norm2 = RMSNorm(dim)

    def forward(self, x: Tensor):
        x = x + self.attn(self.norm1(x))
        x = x + self.mlp(self.norm2(x))
        return x

class GPT(nn.Module):
    def __init__(self, vocab_size: int, num_layers: int, model_dim: int):
        super().__init__()
        self.embed = nn.Embedding(vocab_size, model_dim).bfloat16()
        self.blocks = nn.ModuleList([Block(model_dim) for _ in range(num_layers)])
        self.proj = Linear(model_dim, vocab_size)
        self.norm1 = RMSNorm(model_dim)
        self.norm2 = RMSNorm(model_dim)

    def forward_logits(self, inputs: Tensor) -> Tensor:
        x = self.norm1(self.embed(inputs))
        for block in self.blocks:
            x = block(x)
        logits = self.proj(self.norm2(x)).float()
        logits = 15 * logits * (logits.square() + 15**2).rsqrt()
        return logits

    def forward(self, inputs: Tensor, targets: Tensor | None = None):
        logits = self.forward_logits(inputs)
        if targets is None:
            return logits
        return F.cross_entropy(logits.view(targets.numel(), -1), targets.view(-1), reduction="sum")

_BLOCK_WEIGHT_RE = re.compile(
    r"^blocks\.(\d+)\.(attn\.(?:q|k|v|proj)|mlp\.(?:fc|proj))\.weight$"
)

def _matrix_kind_from_name(name: str) -> str | None:
    m = _BLOCK_WEIGHT_RE.match(name)
    return m.group(2).replace(".", "_") if m else None

def _block_idx_from_name(name: str) -> int | None:
    m = _BLOCK_WEIGHT_RE.match(name)
    return int(m.group(1)) if m else None

ZERO_INIT_MATRIX_KINDS = frozenset(("attn_proj", "mlp_proj"))
ZERO_INIT_TRUST_FLOOR_SCALE = float(os.getenv("MYOPT_ZERO_INIT_TRUST_FLOOR_SCALE", "1.0"))
if ZERO_INIT_TRUST_FLOOR_SCALE < 0.0:
    raise ValueError("MYOPT_ZERO_INIT_TRUST_FLOOR_SCALE must be >= 0")


def _zero_init_trust_floor(p: Tensor, scale: float = ZERO_INIT_TRUST_FLOOR_SCALE) -> Tensor:
    if scale <= 0.0:
        return torch.zeros((), dtype=torch.float32, device=p.device)
    return torch.tensor(scale * math.sqrt(0.33 * p.size(0)), dtype=torch.float32, device=p.device)

@torch.compile
def zeropower_via_newtonschulz5(G: Tensor, steps: int = 12, eps: float = 1e-7) -> Tensor:
    assert G.ndim == 2
    X = G.bfloat16()
    if G.size(0) > G.size(1):
        X = X.T
    X = X / (X.norm() + eps)
    a, b, c = 2, -1.5, 0.5
    for _ in range(steps):
        A = X @ X.T
        B = b * A + c * A @ A
        X = a * X + B @ X
    if G.size(0) > G.size(1):
        X = X.T
    return X.to(G.dtype)


class DistributedMuonLKindwise(torch.optim.Optimizer):
    """
    Muon with three scaling schemes:
      plain     : rho=0, no weight-norm scaling (standard Muon)
      kindwise  : rho=1, xi = EMA(||W||/mean_kind(||W||))
      alllayer  : rho=1, xi = EMA((||W||/||U||)/mean_all(||W||/||U||))
                  with zero-initialized weights left at scale 1, matching
                  kindwise's zero-kind behavior but without an arbitrary norm threshold
      lamb_align: xi = trust_norm/||U|| for each matrix, no type/all-layer mean.
                  For zero-init projection weights,
                  trust_norm=max(||W||, scale * sqrt(0.33 * out_dim)).
      orscale_general:
                  Q = NS_k(mu M + G), no Muon shape factor.
                  D = weight_decay * W + Q, trust_ratio = max(||W||, eps)/(||D||+eps),
                  clipped to [0.5, 1.5], and no decoupled weight decay.
                  Zero-init projection matrices use ratio 1 until ||W|| reaches
                  a fraction of the same-numel non-zero-init minimum ||W||.

    weight_decay_mode:
      decoupled    : apply weight decay outside the aligned update
      wd_in_update : add weight_decay * W into U before alignment/scaling
    """
    def __init__(
        self,
        params,
        lr=0.02,
        weight_decay=0.0,
        momentum=0.95,
        ns_steps=12,
        nesterov=True,
        rho=1.0,
        interval=5,
        param_names=None,
        eps=1e-12,
        scheme="kindwise",
        scale_ema_beta=0.9,
        weight_decay_mode="decoupled",
        orscale_r_min=0.5,
        orscale_r_max=1.5,
        orscale_zero_init_same_scale_gate_fraction=0.0,
    ):
        assert scheme in ("plain", "kindwise", "alllayer", "lamb_align", "orscale_general"), f"unknown scheme {scheme!r}"
        weight_decay_mode = str(weight_decay_mode).strip().lower()
        assert weight_decay_mode in ("decoupled", "wd_in_update"), (
            f"unknown weight_decay_mode {weight_decay_mode!r}"
        )
        if scheme == "lamb_align" and weight_decay_mode != "decoupled":
            raise ValueError("lamb_align uses decoupled weight decay; do not put weight decay into U")
        if float(orscale_r_min) > float(orscale_r_max):
            raise ValueError("orscale_r_min must be <= orscale_r_max")
        if float(orscale_zero_init_same_scale_gate_fraction) < 0.0:
            raise ValueError("orscale_zero_init_same_scale_gate_fraction must be >= 0")
        defaults = dict(
            lr=float(lr), weight_decay=float(weight_decay),
            momentum=float(momentum), ns_steps=max(1, int(ns_steps)),
            nesterov=bool(nesterov), rho=float(rho) if scheme != "plain" else 0.0,
            interval=int(interval), eps=float(eps),
            scheme=scheme, scale_ema_beta=float(scale_ema_beta),
            weight_decay_mode=weight_decay_mode,
            orscale_r_min=float(orscale_r_min),
            orscale_r_max=float(orscale_r_max),
            orscale_zero_init_same_scale_gate_fraction=float(
                orscale_zero_init_same_scale_gate_fraction
            ),
        )
        super().__init__(params, defaults)
        self.global_step = 0
        self.scheme = scheme
        self._param_kind_by_id: dict[int, str] = {}
        self._param_block_by_id: dict[int, int] = {}
        self._param_name_by_id: dict[int, str] = {}

        if param_names is not None:
            name_by_id = (param_names if isinstance(param_names, dict)
                          else {id(p): n for n, p in zip(param_names,
                                [p for g in self.param_groups for p in g["params"]])})
            for group in self.param_groups:
                for p in group["params"]:
                    name = name_by_id.get(id(p))
                    if name is None:
                        continue
                    self._param_name_by_id[id(p)] = name
                    self._param_kind_by_id[id(p)] = _matrix_kind_from_name(name) or f"shape:{tuple(p.shape)}"
                    bidx = _block_idx_from_name(name)
                    if bidx is not None:
                        self._param_block_by_id[id(p)] = bidx

    def _matrix_kind(self, p: Tensor) -> str:
        return self._param_kind_by_id.get(id(p), f"shape:{tuple(p.shape)}")

    def _update_relative_scale_ema(self, p: Tensor, raw_scale: Tensor, beta: float) -> Tensor:
        state = self.state[p]
        raw_scale = raw_scale.to(device=p.device, dtype=torch.float32)
        if beta <= 0.0:
            state["relative_scale_ema"] = raw_scale.clone()
            state["relative_scale_ema_updates"] = 1
            return raw_scale
        if "relative_scale_ema" not in state:
            state["relative_scale_ema"] = torch.zeros((), dtype=torch.float32, device=p.device)
            state["relative_scale_ema_updates"] = 0
        ema = state["relative_scale_ema"]
        ema.mul_(beta).add_(raw_scale, alpha=1.0 - beta)
        updates = int(state.get("relative_scale_ema_updates", 0)) + 1
        state["relative_scale_ema_updates"] = updates
        bias_correction = max(float(state.get("eps", 1e-12)), 1.0 - beta ** updates)
        return ema / bias_correction

    def _ensure_param_state(self, p: Tensor) -> dict:
        state = self.state[p]
        if "momentum_buffer" not in state:
            state["momentum_buffer"] = torch.zeros_like(p)
            state.setdefault("matrix_kind", self._matrix_kind(p))
            state.setdefault("cached_w_norm", torch.ones((), dtype=torch.float32, device=p.device))
            state.setdefault("cached_relative_scale", torch.ones((), dtype=torch.float32, device=p.device))
        return state

    def _add_weight_decay_to_update(self, p: Tensor, update: Tensor, weight_decay: float) -> Tensor:
        if self.defaults["weight_decay_mode"] != "wd_in_update" or weight_decay == 0.0:
            return update
        return update.add(p.detach().to(device=update.device, dtype=update.dtype), alpha=weight_decay)

    def _apply_weight_decay_and_update(
        self,
        p: Tensor,
        update: Tensor,
        lr: float,
        weight_decay: float,
    ) -> None:
        if self.defaults["weight_decay_mode"] == "decoupled" and weight_decay != 0.0:
            p.mul_(1 - lr * weight_decay)
        p.add_(update.reshape_as(p), alpha=-lr)

    @torch.no_grad()
    def _refresh_all_scales(self) -> None:
        """Refresh cached_relative_scale for all matrix params (all groups)."""
        if self.scheme in ("plain", "alllayer", "lamb_align"):
            return

        interval = self.defaults["interval"]
        needs_refresh = (
            any("cached_relative_scale" not in self.state[p]
                for group in self.param_groups
                for p in group["params"] if p.ndim >= 2)
            or self.global_step % interval == 0
        )
        if not needs_refresh:
            return

        all_matrix = [(p, group)
                      for group in self.param_groups
                      for p in group["params"] if p.ndim >= 2]
        eps = self.defaults["eps"]
        beta = self.defaults["scale_ema_beta"]

        norms = {id(p): torch.linalg.vector_norm(p.detach().float()) for p, _ in all_matrix}

        by_kind: dict[str, list] = {}
        for p, _ in all_matrix:
            by_kind.setdefault(self._matrix_kind(p), []).append(p)
        for kind, params_k in by_kind.items():
            avg_norm_unclamped = torch.stack([norms[id(p)] for p in params_k]).mean()
            avg_k = avg_norm_unclamped.clamp_min(eps)
            zero_kind = bool(avg_norm_unclamped <= eps)
            for p in params_k:
                raw = torch.ones_like(norms[id(p)]) if zero_kind else norms[id(p)] / avg_k
                self.state[p]["cached_relative_scale"] = self._update_relative_scale_ema(p, raw, beta)
                self.state[p]["cached_w_norm"] = norms[id(p)]
                self.state[p]["matrix_kind"] = kind

    @torch.no_grad()
    def step(self):
        if self.scheme == "orscale_general":
            self._step_orscale_general()
            return
        if self.scheme == "alllayer":
            self._step_alllayer_ratio_aligned()
            return
        if self.scheme == "lamb_align":
            self._step_lamb_align()
            return

        world_size = dist.get_world_size()
        rank = dist.get_rank()
        self._refresh_all_scales()

        for group in self.param_groups:
            beta = float(group["momentum"])
            ns_steps = max(1, int(group["ns_steps"]))
            nesterov = bool(group["nesterov"])
            rho = float(group["rho"])
            params = group["params"]
            params_pad = params + [torch.empty_like(params[-1])] * (world_size - len(params) % world_size)

            for base_i in range(0, len(params), world_size):
                if base_i + rank < len(params):
                    p = params[base_i + rank]
                    if p.grad is not None:
                        state = self._ensure_param_state(p)
                        grad = p.grad
                        buf = state["momentum_buffer"]
                        buf.lerp_(grad, 1 - beta)
                        update = grad.lerp_(buf, beta) if nesterov else buf
                        state["last_update_buffer_pre_ns"] = update.detach().clone()
                        update = zeropower_via_newtonschulz5(update, steps=ns_steps)
                        update *= max(1, update.size(-2) / update.size(-1)) ** 0.5
                        update = self._add_weight_decay_to_update(
                            p, update, float(group["weight_decay"])
                        )
                        if rho > 0.0:
                            rel = state["cached_relative_scale"].to(dtype=update.dtype)
                            update = update * rel.pow(rho)

                        self._apply_weight_decay_and_update(
                            p,
                            update,
                            float(group["lr"]),
                            float(group["weight_decay"]),
                        )

                dist.all_gather(params_pad[base_i:base_i + world_size], params_pad[base_i + rank])

        self.global_step += 1

    @torch.no_grad()
    def _step_orscale_general(self):
        world_size = dist.get_world_size()
        rank = dist.get_rank()
        eps = float(self.defaults["eps"])
        r_min = float(self.defaults["orscale_r_min"])
        r_max = float(self.defaults["orscale_r_max"])
        gate_fraction = float(self.defaults["orscale_zero_init_same_scale_gate_fraction"])

        w_norm_by_id: dict[int, Tensor] = {}
        same_scale_normal_min: dict[int, Tensor] = {}
        same_scale_normal_count: dict[int, int] = {}
        if gate_fraction > 0.0:
            for group in self.param_groups:
                for p in group["params"]:
                    if p.ndim < 2:
                        continue
                    w_norm = torch.linalg.vector_norm(p.detach().float())
                    w_norm_by_id[id(p)] = w_norm
                    if self._matrix_kind(p) in ZERO_INIT_MATRIX_KINDS:
                        continue
                    if not bool(torch.isfinite(w_norm).item()):
                        continue
                    if float(w_norm.item()) <= eps:
                        continue
                    key = int(p.numel())
                    same_scale_normal_count[key] = same_scale_normal_count.get(key, 0) + 1
                    current = same_scale_normal_min.get(key)
                    same_scale_normal_min[key] = w_norm if current is None else torch.minimum(current, w_norm)

        for group in self.param_groups:
            beta = float(group["momentum"])
            ns_steps = max(1, int(group["ns_steps"]))
            nesterov = bool(group["nesterov"])
            lr = float(group["lr"])
            wd = float(group["weight_decay"])
            params = group["params"]
            params_pad = params + [torch.empty_like(params[-1])] * (world_size - len(params) % world_size)

            for base_i in range(0, len(params), world_size):
                if base_i + rank < len(params):
                    p = params[base_i + rank]
                    if p.grad is not None:
                        state = self._ensure_param_state(p)
                        grad = p.grad
                        buf = state["momentum_buffer"]
                        buf.mul_(beta).add_(grad)
                        update_pre_ns = grad.add(buf, alpha=beta) if nesterov else buf
                        q = zeropower_via_newtonschulz5(update_pre_ns, steps=ns_steps).float()
                        direction = q
                        if wd != 0.0:
                            direction = direction.add(
                                p.detach().to(device=direction.device, dtype=direction.dtype),
                                alpha=wd,
                            )

                        w_norm = w_norm_by_id.get(id(p))
                        if w_norm is None:
                            w_norm = torch.linalg.vector_norm(p.detach().float())
                        direction_norm = torch.linalg.vector_norm(direction.detach().float())
                        raw_ratio = w_norm.clamp_min(eps) / direction_norm.add(eps)
                        clipped_ratio = torch.clamp(raw_ratio, min=r_min, max=r_max)
                        effective_ratio = clipped_ratio

                        kind = self._matrix_kind(p)
                        gate_enabled = bool(gate_fraction > 0.0 and kind in ZERO_INIT_MATRIX_KINDS)
                        gate_open = True
                        gate_status = "standard"
                        gate_threshold = None
                        same_scale_min = None
                        same_scale_count = 0
                        if gate_enabled:
                            key = int(p.numel())
                            same_scale_min = same_scale_normal_min.get(key)
                            same_scale_count = int(same_scale_normal_count.get(key, 0))
                            gate_open = False
                            if same_scale_min is not None:
                                gate_threshold = same_scale_min.to(device=w_norm.device, dtype=torch.float32) * gate_fraction
                                gate_open = bool(float(w_norm.item()) >= float(gate_threshold.item()))
                                gate_status = "same_scale_gate_open" if gate_open else "same_scale_bootstrap_ratio_1"
                            else:
                                gate_status = "same_scale_min_unavailable_ratio_1"
                            if not gate_open:
                                effective_ratio = torch.ones_like(clipped_ratio)

                        state["last_update_buffer_pre_ns"] = update_pre_ns.detach().clone()
                        state["cached_w_norm"] = w_norm.detach().to(device=p.device, dtype=torch.float32)
                        state["cached_relative_scale"] = effective_ratio.detach().to(device=p.device, dtype=torch.float32)
                        state["matrix_kind"] = kind
                        state["orscale_general_q_norm"] = torch.linalg.vector_norm(q.detach().float()).to(
                            device=p.device, dtype=torch.float32
                        )
                        state["orscale_general_direction_norm"] = direction_norm.detach().to(
                            device=p.device, dtype=torch.float32
                        )
                        state["orscale_general_raw_trust_ratio"] = raw_ratio.detach().to(
                            device=p.device, dtype=torch.float32
                        )
                        state["orscale_general_clipped_trust_ratio"] = clipped_ratio.detach().to(
                            device=p.device, dtype=torch.float32
                        )
                        state["orscale_general_effective_trust_ratio"] = effective_ratio.detach().to(
                            device=p.device, dtype=torch.float32
                        )
                        state["orscale_general_zero_init_same_scale_gate_enabled"] = gate_enabled
                        state["orscale_general_zero_init_same_scale_gate_open"] = gate_open
                        state["orscale_general_zero_init_same_scale_gate_status"] = gate_status
                        state["orscale_general_zero_init_same_scale_gate_fraction"] = gate_fraction
                        state["orscale_general_same_scale_normal_w_norm_min"] = (
                            same_scale_min.detach().to(device=p.device, dtype=torch.float32)
                            if same_scale_min is not None else None
                        )
                        state["orscale_general_same_scale_normal_w_norm_count"] = same_scale_count
                        state["orscale_general_w_norm_gate_threshold"] = (
                            gate_threshold.detach().to(device=p.device, dtype=torch.float32)
                            if gate_threshold is not None else None
                        )

                        scaled_direction = direction * effective_ratio.to(device=direction.device, dtype=direction.dtype)
                        p.add_(scaled_direction.reshape_as(p).to(dtype=p.dtype), alpha=-lr)

                dist.all_gather(params_pad[base_i:base_i + world_size], params_pad[base_i + rank])

        self.global_step += 1

    @torch.no_grad()
    def _step_lamb_align(self):
        world_size = dist.get_world_size()
        rank = dist.get_rank()
        eps = float(self.defaults["eps"])

        for group in self.param_groups:
            beta = float(group["momentum"])
            ns_steps = max(1, int(group["ns_steps"]))
            nesterov = bool(group["nesterov"])
            lr = float(group["lr"])
            wd = float(group["weight_decay"])
            params = group["params"]
            params_pad = params + [torch.empty_like(params[-1])] * (world_size - len(params) % world_size)

            for base_i in range(0, len(params), world_size):
                if base_i + rank < len(params):
                    p = params[base_i + rank]
                    if p.grad is not None:
                        state = self._ensure_param_state(p)
                        grad = p.grad
                        buf = state["momentum_buffer"]
                        buf.lerp_(grad, 1 - beta)
                        update = grad.lerp_(buf, beta) if nesterov else buf
                        state["last_update_buffer_pre_ns"] = update.detach().clone()
                        update = zeropower_via_newtonschulz5(update, steps=ns_steps)
                        update *= max(1, update.size(-2) / update.size(-1)) ** 0.5

                        w_norm = torch.linalg.vector_norm(p.detach().float())
                        update_norm = torch.linalg.vector_norm(update.detach().float()).clamp_min(eps)
                        kind = self._matrix_kind(p)
                        trust_floor = None
                        trust_floor_source = "none"
                        trust_norm = w_norm
                        used_trust_floor = False
                        if kind in ZERO_INIT_MATRIX_KINDS:
                            trust_floor = _zero_init_trust_floor(p)
                            trust_floor_source = f"expected_init_norm_x{ZERO_INIT_TRUST_FLOOR_SCALE:g}"
                            trust_norm = torch.maximum(w_norm, trust_floor)
                            used_trust_floor = bool(float(w_norm.item()) < float(trust_floor.item()))
                        if float(trust_norm.item()) > eps:
                            coeff = (trust_norm / update_norm).to(device=p.device, dtype=update.dtype)
                        else:
                            coeff = torch.ones((), dtype=update.dtype, device=p.device)
                        state["cached_w_norm"] = w_norm.detach().to(device=p.device, dtype=torch.float32)
                        state["cached_relative_scale"] = coeff.detach().to(device=p.device, dtype=torch.float32)
                        state["matrix_kind"] = kind
                        state["lamb_align_trust_norm"] = trust_norm.detach().to(device=p.device, dtype=torch.float32)
                        state["lamb_align_used_trust_floor"] = used_trust_floor
                        state["lamb_align_trust_floor"] = (
                            trust_floor.detach().to(device=p.device, dtype=torch.float32)
                            if trust_floor is not None else None
                        )
                        state["lamb_align_trust_floor_source"] = trust_floor_source
                        state["lamb_align_zero_init_trust_floor_scale"] = ZERO_INIT_TRUST_FLOOR_SCALE

                        self._apply_weight_decay_and_update(p, update * coeff, lr, wd)

                dist.all_gather(params_pad[base_i:base_i + world_size], params_pad[base_i + rank])

        self.global_step += 1

    @torch.no_grad()
    def _step_alllayer_ratio_aligned(self):
        world_size = dist.get_world_size()
        rank = dist.get_rank()
        eps = float(self.defaults["eps"])
        interval = int(self.defaults["interval"])
        beta_ema = float(self.defaults["scale_ema_beta"])
        needs_refresh = self.global_step % interval == 0

        first_param = next(p for group in self.param_groups for p in group["params"])
        ratio_sum = torch.zeros((), dtype=torch.float32, device=first_param.device)
        ratio_count = torch.zeros((), dtype=torch.float32, device=first_param.device)
        pending_by_group = []

        for group in self.param_groups:
            beta = float(group["momentum"])
            ns_steps = max(1, int(group["ns_steps"]))
            nesterov = bool(group["nesterov"])
            wd = float(group["weight_decay"])
            params = group["params"]
            params_pad = params + [torch.empty_like(params[-1])] * (world_size - len(params) % world_size)
            pending = []

            for base_i in range(0, len(params), world_size):
                if base_i + rank < len(params):
                    p = params[base_i + rank]
                    if p.grad is not None:
                        state = self._ensure_param_state(p)
                        if "relative_scale_ema" not in state:
                            needs_refresh = True
                        grad = p.grad
                        buf = state["momentum_buffer"]
                        buf.lerp_(grad, 1 - beta)
                        update = grad.lerp_(buf, beta) if nesterov else buf
                        state["last_update_buffer_pre_ns"] = update.detach().clone()
                        update = zeropower_via_newtonschulz5(update, steps=ns_steps)
                        update *= max(1, update.size(-2) / update.size(-1)) ** 0.5
                        update = self._add_weight_decay_to_update(p, update, wd)

                        w_norm = torch.linalg.vector_norm(p.detach().float())
                        update_norm = torch.linalg.vector_norm(update.detach().float()).clamp_min(eps)
                        zero_weight = bool(float(w_norm.item()) <= eps)
                        ratio = None
                        if needs_refresh and not zero_weight:
                            ratio = w_norm / update_norm
                            ratio_sum.add_(ratio.detach().float())
                            ratio_count.add_(1.0)

                        state["cached_w_norm"] = w_norm.detach().to(device=p.device, dtype=torch.float32)
                        state["matrix_kind"] = self._matrix_kind(p)
                        pending.append((p, update, state, ratio))

            pending_by_group.append((group, params, params_pad, pending))

        ratio_mean = None
        zero_all = False
        if needs_refresh:
            dist.all_reduce(ratio_sum, op=dist.ReduceOp.SUM)
            dist.all_reduce(ratio_count, op=dist.ReduceOp.SUM)
            ratio_mean = ratio_sum / ratio_count.clamp_min(1.0)
            zero_all = bool(float(ratio_mean.item()) <= eps)

        for group, params, params_pad, pending in pending_by_group:
            lr = float(group["lr"])
            wd = float(group["weight_decay"])
            rho = float(group["rho"])
            for p, update, state, ratio in pending:
                if needs_refresh:
                    if ratio is None or rho <= 0.0 or zero_all:
                        raw_coeff = torch.ones((), dtype=torch.float32, device=p.device)
                    else:
                        raw_coeff = (ratio / ratio_mean.clamp_min(eps)).pow(rho)
                        raw_coeff = raw_coeff.to(device=p.device, dtype=torch.float32)
                    state["cached_relative_scale"] = self._update_relative_scale_ema(p, raw_coeff, beta_ema)
                coeff = state["cached_relative_scale"].to(device=p.device, dtype=update.dtype)
                self._apply_weight_decay_and_update(p, update * coeff, lr, wd)

            for base_i in range(0, len(params), world_size):
                dist.all_gather(params_pad[base_i:base_i + world_size], params_pad[base_i + rank])

        self.global_step += 1

def _ns5_and_shape_scale(buf: Tensor, ns_steps: int) -> tuple[Tensor, float]:
    """Returns (vi_raw, shape_scale) where vi_raw is the Muon update to subtract."""
    buf_2d = buf.view(buf.shape[0], -1).float()
    vi_2d = zeropower_via_newtonschulz5(buf_2d, steps=ns_steps).float()
    shape_scale = max(1.0, vi_2d.size(-2) / vi_2d.size(-1)) ** 0.5
    vi_raw = vi_2d * shape_scale
    return vi_raw.reshape(buf.shape), shape_scale


def _safe_float(x) -> float:
    v = float(x)
    return v if math.isfinite(v) else None


def _safe_float_or_none(x) -> float | None:
    if x is None:
        return None
    return _safe_float(x)


def _tensor_to_matrix(t: Tensor) -> Tensor:
    if t.ndim < 2:
        raise ValueError(f"expected matrix-like tensor, got shape={tuple(t.shape)}")
    if t.ndim == 2:
        return t
    return t.reshape(t.shape[0], -1)


def _safe_ratio(num: float, den: float) -> float | None:
    if not (math.isfinite(num) and math.isfinite(den)) or abs(den) <= 1e-30:
        return None
    return num / den


def _population_nu(values: list[float]) -> float | None:
    if not values:
        return None
    mean = sum(values) / len(values)
    if abs(mean) <= 1e-30:
        return None
    var = sum((v - mean) ** 2 for v in values) / len(values)
    return var / (mean * mean)


def _pearson_corr(xs: list[float], ys: list[float]) -> float | None:
    if len(xs) != len(ys) or len(xs) < 2:
        return None
    mx = sum(xs) / len(xs)
    my = sum(ys) / len(ys)
    vx = sum((x - mx) ** 2 for x in xs)
    vy = sum((y - my) ** 2 for y in ys)
    den = (vx * vy) ** 0.5
    if den <= 1e-30:
        return None
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / den


def _scheme_orscale_stats(
    *,
    alpha: list[float],
    a_vals: list[float],
    b_vals: list[float],
    z_vals: list[float],
) -> dict:
    y_vals = [al * math.sqrt(max(bi, 0.0)) for al, bi in zip(alpha, b_vals)]
    numerator = sum(al * ai for al, ai in zip(alpha, a_vals))
    denominator = sum(bi * al * al for bi, al in zip(b_vals, alpha))
    phi = (numerator * numerator / denominator) if denominator > 0.0 else None

    yz_dot = sum(z * y for z, y in zip(z_vals, y_vals))
    z_norm_sq = sum(z * z for z in z_vals)
    y_norm_sq = sum(y * y for y in y_vals)
    cos2 = (
        yz_dot * yz_dot / (z_norm_sq * y_norm_sq)
        if z_norm_sq > 0.0 and y_norm_sq > 0.0 else None
    )
    nu = _population_nu(y_vals)
    rho = _pearson_corr(y_vals, z_vals)
    return {
        "Phi": _safe_float(phi) if phi is not None else None,
        "sum_alpha_a": _safe_float(numerator),
        "sum_b_alpha_sq": _safe_float(denominator),
        "cos2": _safe_float(cos2) if cos2 is not None else None,
        "nu": _safe_float(nu) if nu is not None else None,
        "rho": _safe_float(rho) if rho is not None else None,
    }


def _vector_match_stats(xs: list[float], ts: list[float]) -> dict:
    finite_pairs = [
        (float(x), float(t))
        for x, t in zip(xs, ts)
        if math.isfinite(float(x)) and math.isfinite(float(t))
    ]
    if not finite_pairs:
        return {
            "n": 0,
            "cosine": None,
            "pearson": None,
            "best_scale_to_t": None,
            "relative_l2_after_best_scale": None,
            "direct_relative_l2": None,
            "mean_x": None,
            "mean_t": None,
            "std_x": None,
            "std_t": None,
        }

    x_vals = [p[0] for p in finite_pairs]
    t_vals = [p[1] for p in finite_pairs]
    dot_xt = sum(x * t for x, t in finite_pairs)
    dot_xx = sum(x * x for x in x_vals)
    dot_tt = sum(t * t for t in t_vals)
    norm_prod = math.sqrt(max(dot_xx, 0.0) * max(dot_tt, 0.0))
    cosine = dot_xt / norm_prod if norm_prod > 1e-30 else None
    pearson = _pearson_corr(x_vals, t_vals)
    best_scale = dot_xt / dot_xx if dot_xx > 1e-30 else None
    if best_scale is not None and dot_tt > 1e-30:
        rel_l2 = math.sqrt(sum((best_scale * x - t) ** 2 for x, t in finite_pairs) / dot_tt)
    else:
        rel_l2 = None
    direct_rel_l2 = (
        math.sqrt(sum((x - t) ** 2 for x, t in finite_pairs) / dot_tt)
        if dot_tt > 1e-30 else None
    )
    mean_x = sum(x_vals) / len(x_vals)
    mean_t = sum(t_vals) / len(t_vals)
    std_x = math.sqrt(sum((x - mean_x) ** 2 for x in x_vals) / len(x_vals))
    std_t = math.sqrt(sum((t - mean_t) ** 2 for t in t_vals) / len(t_vals))
    return {
        "n": len(finite_pairs),
        "cosine": _safe_float_or_none(cosine),
        "pearson": _safe_float_or_none(pearson),
        "best_scale_to_t": _safe_float_or_none(best_scale),
        "relative_l2_after_best_scale": _safe_float_or_none(rel_l2),
        "direct_relative_l2": _safe_float_or_none(direct_rel_l2),
        "mean_x": _safe_float(mean_x),
        "mean_t": _safe_float(mean_t),
        "std_x": _safe_float(std_x),
        "std_t": _safe_float(std_t),
    }


def _make_block_ggn_operator(
    *,
    model: GPT,
    diag_inputs: Tensor,
    param_name: str,
    num_tokens_diag: int,
    ggn_microbatch_size: int,
):
    """Return H_GGN[delta] for one parameter block using chunked JVP/VJP on logits."""
    params = {name: p.detach().float() for name, p in model.named_parameters()}
    buffers = {
        name: b.detach().float() if b.is_floating_point() else b.detach()
        for name, b in model.named_buffers()
    }
    base_param = params[param_name]
    inv_tokens = 1.0 / float(max(num_tokens_diag, 1))
    chunk_size = max(1, int(ggn_microbatch_size))

    def operator(delta: Tensor) -> Tensor:
        delta = delta.to(device=base_param.device, dtype=base_param.dtype)
        ggn_vec = torch.zeros_like(base_param, dtype=torch.float32)
        for start in range(0, diag_inputs.size(0), chunk_size):
            inputs_chunk = diag_inputs[start:start + chunk_size]

            def logits_from_param(param_value: Tensor) -> Tensor:
                local_params = dict(params)
                local_params[param_name] = param_value
                if inputs_chunk.is_cuda:
                    sdp_ctx = sdpa_kernel(SDPBackend.MATH)
                else:
                    sdp_ctx = nullcontext()
                with sdp_ctx:
                    logits = functional_call(
                        model,
                        (local_params, buffers),
                        args=(inputs_chunk, None),
                        tie_weights=True,
                        strict=False,
                    )
                return logits.float().reshape(-1, logits.shape[-1])

            base_logits_flat, vjp_fn = vjp(logits_from_param, base_param)
            probs = torch.softmax(base_logits_flat.detach().float(), dim=-1)
            _, logits_tangent = jvp(logits_from_param, (base_param,), (delta,))
            tangent = logits_tangent.float().reshape_as(probs)
            centered = tangent - torch.sum(probs * tangent, dim=-1, keepdim=True)
            hz_jv = probs * centered * inv_tokens
            (chunk_vec,) = vjp_fn(hz_jv.to(dtype=base_logits_flat.dtype))
            ggn_vec.add_(chunk_vec.detach().float())
        if dist.is_initialized():
            dist.all_reduce(ggn_vec, op=dist.ReduceOp.SUM)
        return ggn_vec

    return operator


def _exact_block_hessian_vec(
    *,
    model: GPT,
    diag_inputs: Tensor,
    diag_targets: Tensor,
    param: Tensor,
    vector: Tensor,
    num_tokens_diag: int,
    hvp_microbatch_size: int,
) -> Tensor:
    """Return H_ii @ vector for the diagnostic per-token mean loss."""
    hv = torch.zeros_like(param, dtype=torch.float32)
    v = vector.detach().to(device=param.device, dtype=param.dtype)
    chunk_size = max(1, int(hvp_microbatch_size))

    for start in range(0, diag_inputs.size(0), chunk_size):
        inputs_chunk = diag_inputs[start:start + chunk_size]
        targets_chunk = diag_targets[start:start + chunk_size]
        sdp_ctx = sdpa_kernel(SDPBackend.MATH) if inputs_chunk.is_cuda else nullcontext()
        with torch.enable_grad(), sdp_ctx:
            loss_sum = model(inputs_chunk, targets_chunk)
            loss_mean_global = loss_sum / float(max(num_tokens_diag, 1))
            (g,) = torch.autograd.grad(
                loss_mean_global,
                param,
                create_graph=True,
                allow_unused=True,
            )
            if g is None:
                continue
            gv = torch.sum(g * v)
            if not gv.requires_grad:
                continue
            (hv_chunk,) = torch.autograd.grad(gv, param, allow_unused=True)
        if hv_chunk is not None:
            hv.add_(hv_chunk.detach().float())

    if dist.is_initialized():
        dist.all_reduce(hv, op=dist.ReduceOp.SUM)
    return hv


def _exact_source_block_hessian_vecs(
    *,
    model: GPT,
    diag_inputs: Tensor,
    diag_targets: Tensor,
    source_param: Tensor,
    source_vector: Tensor,
    target_params: list[Tensor],
    num_tokens_diag: int,
    hvp_microbatch_size: int,
) -> dict[int, Tensor]:
    """Return {id(target): H_target,source @ source_vector} for the diagnostic loss."""
    hv_by_pid = {
        id(p): torch.zeros_like(p, dtype=torch.float32)
        for p in target_params
    }
    v = source_vector.detach().to(device=source_param.device, dtype=source_param.dtype)
    chunk_size = max(1, int(hvp_microbatch_size))

    for start in range(0, diag_inputs.size(0), chunk_size):
        inputs_chunk = diag_inputs[start:start + chunk_size]
        targets_chunk = diag_targets[start:start + chunk_size]
        sdp_ctx = sdpa_kernel(SDPBackend.MATH) if inputs_chunk.is_cuda else nullcontext()
        with torch.enable_grad(), sdp_ctx:
            loss_sum = model(inputs_chunk, targets_chunk)
            loss_mean_global = loss_sum / float(max(num_tokens_diag, 1))
            (g_source,) = torch.autograd.grad(
                loss_mean_global,
                source_param,
                create_graph=True,
                allow_unused=True,
            )
            if g_source is None:
                continue
            gv = torch.sum(g_source * v)
            if not gv.requires_grad:
                continue
            hv_chunks = torch.autograd.grad(gv, target_params, allow_unused=True)
        for target_param, hv_chunk in zip(target_params, hv_chunks):
            if hv_chunk is not None:
                hv_by_pid[id(target_param)].add_(hv_chunk.detach().float())

    if dist.is_initialized():
        for hv in hv_by_pid.values():
            dist.all_reduce(hv, op=dist.ReduceOp.SUM)
    return hv_by_pid


def _estimate_block_lmax(
    *,
    ggn_operator,
    shape: torch.Size,
    device: torch.device,
    power_iters: int,
    seed: int,
    floor: float,
    initial_vector: Tensor | None = None,
    eps: float = 1e-12,
) -> dict:
    gen = torch.Generator(device=device)
    gen.manual_seed(int(seed))
    if initial_vector is not None and tuple(initial_vector.shape) == tuple(shape):
        u = initial_vector.to(device=device, dtype=torch.float32, non_blocking=True)
        vector_source = "warm_start"
    else:
        u = torch.randn(shape, device=device, dtype=torch.float32, generator=gen)
        vector_source = "random_init"
    u = u / torch.linalg.vector_norm(u).clamp_min(eps)

    last_norm = None
    for _ in range(max(1, int(power_iters))):
        z = ggn_operator(u)
        z_norm = torch.linalg.vector_norm(z)
        last_norm = float(z_norm)
        if not math.isfinite(last_norm) or last_norm <= eps:
            u = torch.randn(shape, device=device, dtype=torch.float32, generator=gen)
            u = u / torch.linalg.vector_norm(u).clamp_min(eps)
        else:
            u = (z / z_norm).detach()

    h_u = ggn_operator(u)
    raw_L = float(torch.sum(u * h_u).item())
    floor_applied = (not math.isfinite(raw_L)) or raw_L <= floor
    L = max(raw_L if math.isfinite(raw_L) else floor, floor)
    return {
        "L_i": L,
        "L_i_raw": raw_L if math.isfinite(raw_L) else None,
        "L_i_floor_applied": bool(floor_applied),
        "power_last_hvp_norm": last_norm,
        "power_vector_cpu": u.detach().cpu(),
        "power_vector_source": vector_source,
    }


def compute_orscale_diagnostics(
    model: GPT,
    optimizer2: DistributedMuonLKindwise,
    diag_inputs: Tensor,
    diag_targets: Tensor,
    train_batch_tokens: int,
    diag_grad_microbatch_size: int,
    lr_current: float,
    diag_index: int,
    step: int,
    train_loss: float,
    scheme: str,
    seed: int,
    ns_steps: int,
    ggn_microbatch_size: int,
    max_blocks: int,
    l_estimator: str,
    curvature_cache: dict[str, dict],
    l_init_power_iters: int,
    l_refresh_diag_interval: int,
    l_refresh_power_iters: int,
    l_blocks_per_diag: int,
    l_floor: float,
    eps_w: float,
) -> dict:
    """
    Compute one OrScale diagnostic record at the current checkpoint.
    Returns a dict ready for JSON serialization.
    Called every DIAG_FREQ steps, before the optimizer step.
    """
    world_size = dist.get_world_size()
    num_tokens_diag = diag_inputs.numel() * world_size  # global diagnostic tokens
    momentum_debug_scale = 1.0 / train_batch_tokens

    # ------------------------------------------------------------------ #
    # 1. Collect exactly the MuonL matrix blocks, excluding AdamW fallback #
    # ------------------------------------------------------------------ #
    block_params = []  # list of (p, name, kind, block_idx, base_lr, raw_wd)
    group_cfg_by_pid = {}
    for group in optimizer2.param_groups:
        base_lr = float(group.get("initial_lr", group["lr"]))
        raw_wd = float(group["weight_decay"])
        beta = float(group["momentum"])
        nesterov = bool(group["nesterov"])
        for p in group["params"]:
            if p.ndim < 2:
                continue
            state = optimizer2.state[p]
            if "momentum_buffer" not in state:
                continue
            name = optimizer2._param_name_by_id.get(id(p), f"id_{id(p)}")
            kind = optimizer2._matrix_kind(p)
            block_idx = optimizer2._param_block_by_id.get(id(p), -1)
            block_params.append((p, name, kind, block_idx, base_lr, raw_wd))
            group_cfg_by_pid[id(p)] = {"momentum": beta, "nesterov": nesterov}
    block_params.sort(key=lambda item: (item[2], item[3], item[1]))
    if max_blocks > 0:
        block_params = block_params[:max_blocks]

    def compute_diag_gradient() -> tuple[float, float, dict[int, Tensor], dict[int, list[Tensor]]]:
        """Return diagnostic gradients plus per-microbatch block gradients."""
        saved_grads = [(p, None if p.grad is None else p.grad.detach().clone())
                       for p in model.parameters()]
        loss_sum_global = torch.zeros((), device=diag_inputs.device, dtype=torch.float32)
        chunk_size = max(1, int(diag_grad_microbatch_size))
        g_diag_accum: dict[int, Tensor] = {}
        microbatch_grads_cpu: dict[int, list[Tensor]] = {id(p): [] for p, *_ in block_params}

        for start in range(0, diag_inputs.size(0), chunk_size):
            model.zero_grad(set_to_none=True)
            inputs_chunk = diag_inputs[start:start + chunk_size]
            targets_chunk = diag_targets[start:start + chunk_size]
            loss_sum_local = model(inputs_chunk, targets_chunk)
            loss_sum_global.add_(loss_sum_local.detach().float())
            chunk_tokens_global = inputs_chunk.numel() * world_size
            chunk_weight = float(chunk_tokens_global) / float(max(num_tokens_diag, 1))
            (loss_sum_local / float(max(chunk_tokens_global, 1))).backward()

            for p, *_ in block_params:
                if p.grad is None:
                    raise RuntimeError(f"missing diagnostic gradient for {optimizer2._param_name_by_id.get(id(p), id(p))}")
                dist.all_reduce(p.grad, op=dist.ReduceOp.SUM)
                g_chunk = p.grad.detach().float().clone()
                pid = id(p)
                if pid not in g_diag_accum:
                    g_diag_accum[pid] = g_chunk.mul(chunk_weight)
                else:
                    g_diag_accum[pid].add_(g_chunk, alpha=chunk_weight)
                if l_estimator == "microbatch_fisher_gram":
                    microbatch_grads_cpu[pid].append(g_chunk.reshape(-1).cpu())

        dist.all_reduce(loss_sum_global, op=dist.ReduceOp.SUM)

        g_diag: dict[int, Tensor] = {}
        for p, *_ in block_params:
            g_diag[id(p)] = g_diag_accum[id(p)]

        model.zero_grad(set_to_none=True)
        for p, g in saved_grads:
            p.grad = g
        f_sum = float(loss_sum_global)
        return f_sum, f_sum / num_tokens_diag, g_diag, microbatch_grads_cpu

    # ------------------------------------------------------------------ #
    # 2. Diagnostic-batch gradient G_i and nuclear norm a_i               #
    # ------------------------------------------------------------------ #
    f_0_sum, f_0, g_diag, microbatch_grads_cpu = compute_diag_gradient()

    # ------------------------------------------------------------------ #
    # 3. Reconstruct original normalized-direction setup for r_i and x_i  #
    # ------------------------------------------------------------------ #
    block_info: dict[int, dict] = {}
    for p, name, kind, block_idx, _lr, _wd in block_params:
        state = optimizer2.state[p]
        if p.grad is None:
            raise RuntimeError(f"missing current training gradient for {name}")
        beta = group_cfg_by_pid[id(p)]["momentum"]
        nesterov = group_cfg_by_pid[id(p)]["nesterov"]
        train_grad = p.grad.detach().float()
        buf_old = state["momentum_buffer"].detach().float()
        buf_next = torch.lerp(buf_old, train_grad, 1.0 - beta)
        update_buffer = torch.lerp(train_grad, buf_next, beta) if nesterov else buf_next
        vi_raw, shape_scale = _ns5_and_shape_scale(update_buffer, ns_steps)
        vi_raw_norm = float(torch.linalg.vector_norm(vi_raw))
        w = p.detach().float()
        w_norm = float(torch.linalg.vector_norm(w))
        m_i, n_i = int(_tensor_to_matrix(w).shape[0]), int(_tensor_to_matrix(w).shape[1])
        d_i = min(m_i, n_i)
        sqrt_d = math.sqrt(float(d_i))
        r_i = vi_raw_norm / max(w_norm + eps_w, 1e-30)

        g = g_diag[id(p)].float()
        g_mat = _tensor_to_matrix(g)
        a_i = float(torch.linalg.svdvals(g_mat).sum().item())
        block_info[id(p)] = {
            "param_name": name,
            "type": kind,
            "block_idx": block_idx,
            "m_i": m_i,
            "n_i": n_i,
            "d_i": d_i,
            "sqrt_d_i": sqrt_d,
            "w_norm": w_norm,
            "s_i": r_i * w_norm / max(sqrt_d, 1e-30),
            "r_i": r_i,
            "shape_scale": shape_scale,
            "vi_raw_norm": vi_raw_norm,
            "vi_source": "current_train_grad_nesterov_pre_ns",
            "a_i": a_i,
            "gi_frob_norm": float(torch.linalg.vector_norm(g).item()),
            "base_lr": _lr,
            "weight_decay_raw": _wd,
        }

    # ------------------------------------------------------------------ #
    # 4. Compute rbar_C and rbar_all using the arithmetic mean rule       #
    # ------------------------------------------------------------------ #
    by_kind: dict[str, list] = {}
    for p, name, kind, block_idx, _lr, _wd in block_params:
        by_kind.setdefault(kind, []).append(id(p))

    r_bar_C: dict[str, float] = {}
    for kind, pids in by_kind.items():
        r_bar_C[kind] = sum(block_info[pid]["r_i"] for pid in pids) / len(pids)

    r_bar_all = (
        sum(info["r_i"] for info in block_info.values()) / max(len(block_info), 1)
    )

    # ------------------------------------------------------------------ #
    # 5. Get L_i from cache or warm-started blockwise GGN power iteration #
    # ------------------------------------------------------------------ #
    refresh_candidates: list[dict] = []
    l_floor_events = []
    curvature_refresh_count = 0
    curvature_cache_hit_count = 0
    curvature_pending_l_count = 0

    if l_estimator == "microbatch_fisher_gram":
        for p, name, kind, block_idx, _lr, _wd in block_params:
            info = block_info[id(p)]
            micro_grads = microbatch_grads_cpu.get(id(p), [])
            if micro_grads:
                stacked = torch.stack(micro_grads, dim=0).float()
                gram = (stacked @ stacked.T) / float(len(micro_grads))
                eigvals = torch.linalg.eigvalsh(gram)
                l_raw = float(eigvals[-1].item())
                gram_trace = float(torch.trace(gram).item())
            else:
                l_raw = None
                gram_trace = None
            l_floor_applied = (l_raw is None) or (not math.isfinite(l_raw)) or l_raw <= l_floor
            l_i = max(l_raw if (l_raw is not None and math.isfinite(l_raw)) else l_floor, l_floor)
            b_i = l_i * info["d_i"]
            info.update({
                "L_i": l_i,
                "L_i_raw": l_raw if (l_raw is not None and math.isfinite(l_raw)) else None,
                "L_i_floor_applied": bool(l_floor_applied),
                "L_i_microbatch_count": len(micro_grads),
                "L_i_gram_trace": gram_trace,
                "power_last_hvp_norm": None,
                "L_i_refresh": False,
                "L_i_source": "microbatch_empirical_fisher_gram",
                "L_i_age": 0,
                "L_i_cache_diag_index": int(diag_index),
                "L_i_power_iters_this_step": 0,
                "power_vector_source": None,
                "b_i": b_i,
                "b_i_method": "microbatch_empirical_fisher_gram_lmax",
            })
            if l_floor_applied:
                l_floor_events.append({
                    "param_name": name,
                    "L_i_raw": l_raw if (l_raw is not None and math.isfinite(l_raw)) else None,
                    "L_i_floor": l_floor,
                })

    elif l_estimator == "rank1_fisher":
        for p, name, kind, block_idx, _lr, _wd in block_params:
            info = block_info[id(p)]
            l_raw = info["gi_frob_norm"] ** 2
            l_floor_applied = (not math.isfinite(l_raw)) or l_raw <= l_floor
            l_i = max(l_raw if math.isfinite(l_raw) else l_floor, l_floor)
            b_i = l_i * info["d_i"]
            info.update({
                "L_i": l_i,
                "L_i_raw": l_raw if math.isfinite(l_raw) else None,
                "L_i_floor_applied": bool(l_floor_applied),
                "L_i_microbatch_count": None,
                "L_i_gram_trace": None,
                "power_last_hvp_norm": None,
                "L_i_refresh": False,
                "L_i_source": "rank1_batch_fisher_proxy",
                "L_i_age": 0,
                "L_i_cache_diag_index": int(diag_index),
                "L_i_power_iters_this_step": 0,
                "power_vector_source": None,
                "b_i": b_i,
                "b_i_method": "rank1_batch_fisher_proxy_g_frob_sq",
            })
            if l_floor_applied:
                l_floor_events.append({
                    "param_name": name,
                    "L_i_raw": l_raw if math.isfinite(l_raw) else None,
                    "L_i_floor": l_floor,
                })

    elif l_estimator != "ggn":
        raise ValueError(f"unknown MYOPT_ORSCALE_L_ESTIMATOR={l_estimator!r}")

    for block_ord, (p, name, kind, block_idx, _lr, _wd) in enumerate(block_params):
        if l_estimator != "ggn":
            continue
        cache_entry = curvature_cache.get(name)
        if cache_entry is None:
            refresh_candidates.append({
                "block_ord": block_ord,
                "name": name,
                "source": "init_power_iteration",
                "age": None,
                "power_iters": max(1, int(l_init_power_iters)),
                "initial_vector": None,
            })
            continue

        cached_at = int(cache_entry.get("diag_index", 0))
        age = max(0, int(diag_index) - cached_at)
        if int(l_refresh_diag_interval) > 0 and age >= int(l_refresh_diag_interval):
            refresh_candidates.append({
                "block_ord": block_ord,
                "name": name,
                "source": "warm_start_refresh",
                "age": age,
                "power_iters": max(1, int(l_refresh_power_iters)),
                "initial_vector": cache_entry.get("power_vector_cpu"),
            })

    max_refresh = int(l_blocks_per_diag)
    if max_refresh > 0:
        refresh_selected_names = {
            item["name"] for item in refresh_candidates[:max_refresh]
        }
    else:
        refresh_selected_names = {item["name"] for item in refresh_candidates}
    refresh_meta_by_name = {item["name"]: item for item in refresh_candidates}

    for block_ord, (p, name, kind, block_idx, _lr, _wd) in enumerate(block_params):
        if l_estimator != "ggn":
            continue
        cache_entry = curvature_cache.get(name)
        refresh_meta = refresh_meta_by_name.get(name)
        refresh_l_i = name in refresh_selected_names

        if refresh_l_i:
            l_i_source = refresh_meta["source"]
            power_iters_this_step = int(refresh_meta["power_iters"])
            l_i_age = refresh_meta["age"]
            initial_vector = refresh_meta["initial_vector"]
        elif cache_entry is None:
            l_i_source = "pending_l_initialization"
            power_iters_this_step = 0
            l_i_age = None
            initial_vector = None
        else:
            cached_at = int(cache_entry.get("diag_index", 0))
            l_i_age = max(0, int(diag_index) - cached_at)
            l_i_source = "cache_stale_pending_refresh" if refresh_meta is not None else "cache"
            power_iters_this_step = 0
            initial_vector = None

        if refresh_l_i and dist.get_rank() == 0:
            print0(
                f"orscale_diag step:{step} block:{block_ord + 1}/{len(block_params)}"
                f" {name} L_refresh:{l_i_source} iters:{power_iters_this_step}",
                console=True,
                log=False,
            )

        if refresh_l_i:
            ggn_operator = _make_block_ggn_operator(
                model=model,
                diag_inputs=diag_inputs,
                param_name=name,
                num_tokens_diag=num_tokens_diag,
                ggn_microbatch_size=ggn_microbatch_size,
            )
            l_info = _estimate_block_lmax(
                ggn_operator=ggn_operator,
                shape=p.shape,
                device=p.device,
                power_iters=power_iters_this_step,
                seed=seed * 1_000_003 + step * 9_176 + block_ord,
                floor=l_floor,
                initial_vector=initial_vector,
            )
            curvature_cache[name] = {
                "diag_index": int(diag_index),
                "step": int(step),
                "L_i": l_info["L_i"],
                "L_i_raw": l_info["L_i_raw"],
                "L_i_floor_applied": l_info["L_i_floor_applied"],
                "L_i_microbatch_count": None,
                "L_i_gram_trace": None,
                "power_last_hvp_norm": l_info["power_last_hvp_norm"],
                "power_vector_cpu": l_info["power_vector_cpu"],
            }
            curvature_refresh_count += 1
            if torch.cuda.is_available():
                torch.cuda.empty_cache()
        elif cache_entry is None:
            l_info = {
                "L_i": None,
                "L_i_raw": None,
                "L_i_floor_applied": False,
                "power_last_hvp_norm": None,
                "power_vector_source": None,
            }
            curvature_pending_l_count += 1
        else:
            l_info = {
                "L_i": cache_entry["L_i"],
                "L_i_raw": cache_entry.get("L_i_raw"),
                "L_i_floor_applied": bool(cache_entry.get("L_i_floor_applied", False)),
                "L_i_microbatch_count": None,
                "L_i_gram_trace": None,
                "power_last_hvp_norm": cache_entry.get("power_last_hvp_norm"),
                "power_vector_source": "cached_not_used",
            }
            curvature_cache_hit_count += 1

        info = block_info[id(p)]
        b_i = l_info["L_i"] * info["d_i"] if l_info["L_i"] is not None else None
        info.update({
            "L_i": l_info["L_i"],
            "L_i_raw": l_info["L_i_raw"],
            "L_i_floor_applied": l_info["L_i_floor_applied"],
            "power_last_hvp_norm": l_info["power_last_hvp_norm"],
            "L_i_refresh": bool(refresh_l_i),
            "L_i_source": l_i_source,
            "L_i_age": l_i_age,
            "L_i_cache_diag_index": (
                int(diag_index) if refresh_l_i
                else (int(cache_entry.get("diag_index", 0)) if cache_entry is not None else None)
            ),
            "L_i_power_iters_this_step": int(power_iters_this_step),
            "power_vector_source": l_info.get("power_vector_source"),
            "b_i": b_i,
            "b_i_method": (
                "block_ggn_power_iteration_jvp_vjp"
                if b_i is not None else "pending_l_cache_not_used_in_phi"
            ),
        })
        if refresh_l_i and l_info["L_i_floor_applied"]:
            l_floor_events.append({
                "param_name": name,
                "L_i_raw": l_info["L_i_raw"],
                "L_i_floor": l_floor,
            })

    # ------------------------------------------------------------------ #
    # 6. Convert x_i to OrScale alpha_i and compute Phi/geometric stats   #
    # ------------------------------------------------------------------ #
    blocks_out = []
    alpha_plain = []
    alpha_typed = []
    alpha_all = []
    a_vals = []
    b_vals = []
    z_vals = []

    for p, name, kind, block_idx, _lr, _wd in block_params:
        pid = id(p)
        info = block_info[pid]
        sqrt_d = info["sqrt_d_i"]
        w_norm = info["w_norm"]
        r_i = info["r_i"]
        rc = r_bar_C[kind]

        a_plain = r_i * w_norm / max(sqrt_d, 1e-30)
        a_typed = rc * w_norm / max(sqrt_d, 1e-30)
        a_all = r_bar_all * w_norm / max(sqrt_d, 1e-30)
        used_in_phi = info["b_i"] is not None
        sqrt_b = math.sqrt(max(info["b_i"], 0.0)) if used_in_phi else None
        z_i = info["a_i"] / max(sqrt_b, 1e-30) if used_in_phi else None

        if used_in_phi:
            alpha_plain.append(a_plain)
            alpha_typed.append(a_typed)
            alpha_all.append(a_all)
            a_vals.append(info["a_i"])
            b_vals.append(info["b_i"])
            z_vals.append(z_i)

        blocks_out.append({
            "block_name": name,
            "type": kind,
            "block_idx": block_idx,
            "used_in_phi": bool(used_in_phi),
            "m_i": info["m_i"],
            "n_i": info["n_i"],
            "d_i": info["d_i"],
            "w_norm": _safe_float(w_norm),
            "s_i": _safe_float(info["s_i"]),
            "r_i": _safe_float(r_i),
            "rbar_C": _safe_float(rc),
            "rbar_all": _safe_float(r_bar_all),
            "alpha_i_plain": _safe_float(a_plain),
            "alpha_i_typed": _safe_float(a_typed),
            "alpha_i_all": _safe_float(a_all),
            "a_i": _safe_float(info["a_i"]),
            "a_i_definition": "nuclear_norm_of_diagnostic_gradient",
            "L_i": _safe_float_or_none(info["L_i"]),
            "L_i_raw": _safe_float_or_none(info["L_i_raw"]),
            "L_i_floor_applied": bool(info["L_i_floor_applied"]),
            "L_i_microbatch_count": info.get("L_i_microbatch_count"),
            "L_i_gram_trace": _safe_float_or_none(info.get("L_i_gram_trace")),
            "L_i_refresh": bool(info["L_i_refresh"]),
            "L_i_source": info["L_i_source"],
            "L_i_age": info["L_i_age"],
            "L_i_cache_diag_index": (
                int(info["L_i_cache_diag_index"])
                if info["L_i_cache_diag_index"] is not None else None
            ),
            "L_i_power_iters_this_step": int(info["L_i_power_iters_this_step"]),
            "power_vector_source": info["power_vector_source"],
            "power_last_hvp_norm": _safe_float_or_none(info["power_last_hvp_norm"]),
            "b_i": _safe_float_or_none(info["b_i"]),
            "b_i_method": info["b_i_method"],
            "y_i_plain": _safe_float_or_none(a_plain * sqrt_b if sqrt_b is not None else None),
            "y_i_typed": _safe_float_or_none(a_typed * sqrt_b if sqrt_b is not None else None),
            "y_i_all": _safe_float_or_none(a_all * sqrt_b if sqrt_b is not None else None),
            "z_i": _safe_float_or_none(z_i),
            "gradient_source": "diagnostic_gradient_per_token_loss",
            "gi_frob_norm": _safe_float(info["gi_frob_norm"]),
            "shape_scale": _safe_float(info["shape_scale"]),
            "vi_raw_norm": _safe_float(info["vi_raw_norm"]),
            "vi_source": info["vi_source"],
            "base_lr": _safe_float(info["base_lr"]),
            "weight_decay_raw": _safe_float(info["weight_decay_raw"]),
            "weight_decay_included_in_phi": False,
        })

    plain_stats = _scheme_orscale_stats(
        alpha=alpha_plain, a_vals=a_vals, b_vals=b_vals, z_vals=z_vals,
    )
    typed_stats = _scheme_orscale_stats(
        alpha=alpha_typed, a_vals=a_vals, b_vals=b_vals, z_vals=z_vals,
    )
    all_stats = _scheme_orscale_stats(
        alpha=alpha_all, a_vals=a_vals, b_vals=b_vals, z_vals=z_vals,
    )

    groups_data = {}
    for kind, pids in by_kind.items():
        phi_pids = [pid for pid in pids if block_info[pid].get("b_i") is not None]
        groups_data[kind] = {
            "rbar_C": _safe_float(r_bar_C[kind]),
            "num_blocks": len(pids),
            "phi_block_count": len(phi_pids),
            "pending_l_blocks": len(pids) - len(phi_pids),
            "sum_a_i": _safe_float(sum(block_info[pid]["a_i"] for pid in pids)),
            "sum_b_i": _safe_float_or_none(
                sum(block_info[pid]["b_i"] for pid in phi_pids)
                if phi_pids else None
            ),
            "mean_r_i": _safe_float(sum(block_info[pid]["r_i"] for pid in pids) / len(pids)),
            "block_names": [
                block_info[pid]["param_name"]
                for pid in sorted(pids, key=lambda pid: block_info[pid]["block_idx"])
            ],
        }

    phi_plain = plain_stats["Phi"]
    phi_typed = typed_stats["Phi"]
    phi_all = all_stats["Phi"]
    kappa_typed_plain = _safe_ratio(phi_typed, phi_plain) if phi_typed is not None and phi_plain is not None else None
    kappa_all_plain = _safe_ratio(phi_all, phi_plain) if phi_all is not None and phi_plain is not None else None
    kappa_typed_all = _safe_ratio(phi_typed, phi_all) if phi_typed is not None and phi_all is not None else None

    return {
        "diagnostic": "orscale_layer_adaptive_score",
        "diag_index": int(diag_index),
        "step": step,
        "train_loss": _safe_float(train_loss),
        "diag_loss": _safe_float(f_0),
        "diag_loss_sum": _safe_float(f_0_sum),
        "lr_current": _safe_float(lr_current),
        "optimizer": scheme,
        "seed": seed,
        "gradient_source": "diagnostic_gradient_per_token_loss",
        "loss_reduction": "per_token_mean",
        "momentum_signal_scale": _safe_float(momentum_debug_scale),
        "momentum_debug_scale": _safe_float(momentum_debug_scale),
        "diag_grad_microbatch_size": int(diag_grad_microbatch_size),
        "included_block_count": len(blocks_out),
        "phi_block_count": len(a_vals),
        "pending_l_block_count": int(curvature_pending_l_count),
        "phi_scope": (
            "full_initialized_l_cache"
            if len(a_vals) == len(blocks_out) else
            "initialized_l_cache_subset"
        ),
        "included_block_rule": (
            "MuonL matrix blocks in optimizer2 only; AdamW fallback params excluded"
            if max_blocks <= 0 else
            f"debug subset: first {max_blocks} MuonL matrix blocks after deterministic ordering"
        ),
        "direction_normalization": {
            "V_i": "Q_i / ||Q_i||_F",
            "Q_i_OrScale": "sqrt(d_i) * V_i",
            "alpha_i": "x_i * ||W_i||_F / sqrt(d_i)",
        },
        "curvature": {
            "estimator": l_estimator,
            "operator": (
                "microbatch_empirical_fisher_gram_lmax"
                if l_estimator == "microbatch_fisher_gram" else
                "rank1_batch_fisher_proxy_g_frob_sq"
                if l_estimator == "rank1_fisher"
                else "block_generalized_gauss_newton_jvp_vjp"
            ),
            "cache_enabled": bool(l_estimator == "ggn"),
            "init_power_iters": int(l_init_power_iters),
            "refresh_diag_interval": int(l_refresh_diag_interval),
            "refresh_power_iters": int(l_refresh_power_iters),
            "max_blocks_refreshed_per_diag": int(l_blocks_per_diag),
            "refresh_candidates_this_diag": len(refresh_candidates),
            "refreshed_blocks_this_diag": int(curvature_refresh_count),
            "cached_blocks_this_diag": int(curvature_cache_hit_count),
            "pending_l_blocks_this_diag": int(curvature_pending_l_count),
            "microbatch_size": int(ggn_microbatch_size),
            "L_i_floor": _safe_float(l_floor),
            "floor_events": l_floor_events,
        },
        "weight_decay_handling": "decoupled weight decay is omitted from OrScale Phi",
        "rbar_all": _safe_float(r_bar_all),
        "Phi_plain": phi_plain,
        "Phi_typed": phi_typed,
        "Phi_all": phi_all,
        "kappa_typed_plain": _safe_float(kappa_typed_plain) if kappa_typed_plain is not None else None,
        "kappa_all_plain": _safe_float(kappa_all_plain) if kappa_all_plain is not None else None,
        "kappa_typed_all": _safe_float(kappa_typed_all) if kappa_typed_all is not None else None,
        "cos2_plain": plain_stats["cos2"],
        "cos2_typed": typed_stats["cos2"],
        "cos2_all": all_stats["cos2"],
        "nu_plain": plain_stats["nu"],
        "nu_typed": typed_stats["nu"],
        "nu_all": all_stats["nu"],
        "rho_plain": plain_stats["rho"],
        "rho_typed": typed_stats["rho"],
        "rho_all": all_stats["rho"],
        "plain": plain_stats,
        "typed": typed_stats,
        "all": all_stats,
        "ordering": {
            "typed_gt_plain": bool(phi_typed is not None and phi_plain is not None and phi_typed > phi_plain),
            "plain_gt_all": bool(phi_plain is not None and phi_all is not None and phi_plain > phi_all),
            "typed_gt_all": bool(phi_typed is not None and phi_all is not None and phi_typed > phi_all),
            "target_typed_gt_plain_gt_all": bool(
                phi_typed is not None and phi_plain is not None and phi_all is not None
                and phi_typed > phi_plain > phi_all
            ),
        },
        "groups": groups_data,
        "blocks": blocks_out,
    }


def compute_local_quadratic_diagnostics(
    model: GPT,
    optimizer2: DistributedMuonLKindwise,
    diag_inputs: Tensor,
    diag_targets: Tensor,
    diag_data_source: str,
    train_batch_tokens: int,
    diag_grad_microbatch_size: int,
    hvp_microbatch_size: int,
    lr_current: float,
    diag_index: int,
    step: int,
    train_loss: float,
    scheme: str,
    seed: int,
    ns_steps: int,
    max_blocks: int,
    curvature_method: str,
    fd_probes: int,
    fd_rel_eps: float,
    fd_abs_eps: float,
    eps_w: float,
    eps_den: float,
    exact_hvp_eager_diagnostic_model: bool,
    measure_cross_curvature: bool,
    save_cross_curvature_matrix: bool,
) -> dict:
    """
    Measure how well three scalar choices x_i match

        t_i = <G_i, V_i> / (||W_i||_F * <V_i, H_ii V_i>)

    for the basic Muon shape-scaled update

        -eta * r_i * ||W_i||_F * V_i.

    Called every DIAG_FREQ steps, before optimizer.step().
    """
    if torch.cuda.is_available():
        torch.cuda.reset_peak_memory_stats()
        cuda_allocated_start = int(torch.cuda.memory_allocated())
        cuda_reserved_start = int(torch.cuda.memory_reserved())
    else:
        cuda_allocated_start = None
        cuda_reserved_start = None

    world_size = dist.get_world_size()
    num_tokens_diag = diag_inputs.numel() * world_size
    momentum_debug_scale = 1.0 / train_batch_tokens

    block_params = []
    group_cfg_by_pid = {}
    for group in optimizer2.param_groups:
        base_lr = float(group.get("initial_lr", group["lr"]))
        raw_wd = float(group["weight_decay"])
        beta = float(group["momentum"])
        nesterov = bool(group["nesterov"])
        for p in group["params"]:
            if p.ndim < 2:
                continue
            state = optimizer2.state[p]
            if "momentum_buffer" not in state:
                continue
            name = optimizer2._param_name_by_id.get(id(p), f"id_{id(p)}")
            kind = optimizer2._matrix_kind(p)
            block_idx = optimizer2._param_block_by_id.get(id(p), -1)
            block_params.append((p, name, kind, block_idx, base_lr, raw_wd))
            group_cfg_by_pid[id(p)] = {"momentum": beta, "nesterov": nesterov}
    block_params.sort(key=lambda item: (item[3], item[1]))
    if max_blocks > 0:
        block_params = block_params[:max_blocks]

    def compute_diag_gradient() -> tuple[float, float, dict[int, Tensor]]:
        saved_grads = [(p, None if p.grad is None else p.grad.detach().clone())
                       for p in model.parameters()]
        loss_sum_global = torch.zeros((), device=diag_inputs.device, dtype=torch.float32)
        chunk_size = max(1, int(diag_grad_microbatch_size))
        g_diag_accum: dict[int, Tensor] = {}

        for start in range(0, diag_inputs.size(0), chunk_size):
            model.zero_grad(set_to_none=True)
            inputs_chunk = diag_inputs[start:start + chunk_size]
            targets_chunk = diag_targets[start:start + chunk_size]
            loss_sum_local = model(inputs_chunk, targets_chunk)
            loss_sum_global.add_(loss_sum_local.detach().float())
            chunk_tokens_global = inputs_chunk.numel() * world_size
            chunk_weight = float(chunk_tokens_global) / float(max(num_tokens_diag, 1))
            (loss_sum_local / float(max(chunk_tokens_global, 1))).backward()

            for p, *_ in block_params:
                if p.grad is None:
                    raise RuntimeError(f"missing diagnostic gradient for {optimizer2._param_name_by_id.get(id(p), id(p))}")
                dist.all_reduce(p.grad, op=dist.ReduceOp.SUM)
                g_chunk = p.grad.detach().float().clone()
                pid = id(p)
                if pid not in g_diag_accum:
                    g_diag_accum[pid] = g_chunk.mul(chunk_weight)
                else:
                    g_diag_accum[pid].add_(g_chunk, alpha=chunk_weight)

        dist.all_reduce(loss_sum_global, op=dist.ReduceOp.SUM)

        g_diag: dict[int, Tensor] = {}
        for p, *_ in block_params:
            g_diag[id(p)] = g_diag_accum[id(p)]

        model.zero_grad(set_to_none=True)
        for p, g in saved_grads:
            p.grad = g
        f_sum = float(loss_sum_global)
        return f_sum, f_sum / num_tokens_diag, g_diag

    def compute_diag_loss_only() -> tuple[float, float]:
        loss_sum_global = torch.zeros((), device=diag_inputs.device, dtype=torch.float32)
        chunk_size = max(1, int(hvp_microbatch_size))
        was_training = model.training
        model.eval()
        with torch.no_grad():
            for start in range(0, diag_inputs.size(0), chunk_size):
                inputs_chunk = diag_inputs[start:start + chunk_size]
                targets_chunk = diag_targets[start:start + chunk_size]
                loss_sum_local = model(inputs_chunk, targets_chunk)
                loss_sum_global.add_(loss_sum_local.detach().float())
        if was_training:
            model.train()
        dist.all_reduce(loss_sum_global, op=dist.ReduceOp.SUM)
        f_sum = float(loss_sum_global)
        return f_sum, f_sum / num_tokens_diag

    f_0_sum, f_0, g_diag = compute_diag_gradient()

    block_info: dict[int, dict] = {}
    by_kind: dict[str, list[int]] = {}
    for block_ord, (p, name, kind, block_idx, _lr, _wd) in enumerate(block_params):
        state = optimizer2.state[p]
        if p.grad is None:
            raise RuntimeError(f"missing current training gradient for {name}")

        beta = group_cfg_by_pid[id(p)]["momentum"]
        nesterov = group_cfg_by_pid[id(p)]["nesterov"]
        train_grad = p.grad.detach().float()
        buf_old = state["momentum_buffer"].detach().float()
        buf_next = torch.lerp(buf_old, train_grad, 1.0 - beta)
        update_buffer = torch.lerp(train_grad, buf_next, beta) if nesterov else buf_next
        vi_raw, shape_scale = _ns5_and_shape_scale(update_buffer, ns_steps)
        vi_raw = vi_raw.float()
        vi_raw_norm = float(torch.linalg.vector_norm(vi_raw).item())
        if vi_raw_norm <= eps_den or not math.isfinite(vi_raw_norm):
            v_i = torch.zeros_like(vi_raw, dtype=torch.float32)
        else:
            v_i = vi_raw / vi_raw_norm

        w = p.detach().float()
        w_norm = float(torch.linalg.vector_norm(w).item())
        m_i, n_i = int(_tensor_to_matrix(w).shape[0]), int(_tensor_to_matrix(w).shape[1])
        d_i = min(m_i, n_i)
        r_i = vi_raw_norm / max(w_norm + eps_w, 1e-30)

        g_i = g_diag[id(p)].float()
        g_dot_v = float(torch.sum(g_i * v_i).item())
        fd_epsilon = max(float(fd_abs_eps), float(fd_rel_eps) * max(w_norm, 1.0))

        block_info[id(p)] = {
            "param_name": name,
            "type": kind,
            "block_idx": block_idx,
            "m_i": m_i,
            "n_i": n_i,
            "d_i": d_i,
            "w_norm": w_norm,
            "r_i": r_i,
            "shape_scale": shape_scale,
            "vi_raw_norm": vi_raw_norm,
            "vi_unit_norm": float(torch.linalg.vector_norm(v_i).item()),
            "train_grad_frob_norm": float(torch.linalg.vector_norm(train_grad).item()),
            "diag_g_frob_norm": float(torch.linalg.vector_norm(g_i).item()),
            "g_dot_v": g_dot_v,
            "v_i_tensor": v_i,
            "fd_epsilon": fd_epsilon,
            "base_lr": _lr,
            "weight_decay_raw": _wd,
        }
        by_kind.setdefault(kind, []).append(id(p))

    fd_probe_losses = []

    cross_curvature_matrix = None
    cross_curvature_order = None
    if measure_cross_curvature and curvature_method != "exact_hvp":
        raise ValueError("cross curvature measurement currently requires MYOPT_LOCAL_QUADRATIC_CURVATURE=exact_hvp")

    if curvature_method == "exact_hvp" and measure_cross_curvature:
        n_blocks = len(block_params)
        target_params = [p for p, *_ in block_params]
        cross_curvature_matrix = [
            [float("nan") for _ in range(n_blocks)]
            for _ in range(n_blocks)
        ]
        cross_curvature_order = [
            {
                "block_name": name,
                "type": kind,
                "block_idx": int(block_idx),
            }
            for p, name, kind, block_idx, _lr, _wd in block_params
        ]
        for source_ord, (source_p, source_name, _source_kind, _source_block_idx, _lr, _wd) in enumerate(block_params):
            source_info = block_info[id(source_p)]
            source_v = source_info["v_i_tensor"]
            if dist.get_rank() == 0 and n_blocks > 8 and source_ord % 12 == 0:
                print0(
                    f"local_quadratic_diag step:{step} cross_hvp_source:{source_ord + 1}/{n_blocks} {source_name}",
                    console=True,
                    log=False,
                )
            hv_by_pid = _exact_source_block_hessian_vecs(
                model=model,
                diag_inputs=diag_inputs,
                diag_targets=diag_targets,
                source_param=source_p,
                source_vector=source_v,
                target_params=target_params,
                num_tokens_diag=num_tokens_diag,
                hvp_microbatch_size=hvp_microbatch_size,
            )
            for target_ord, (target_p, *_target_rest) in enumerate(block_params):
                target_info = block_info[id(target_p)]
                hv_target = hv_by_pid[id(target_p)]
                cij = float(torch.sum(target_info["v_i_tensor"] * hv_target).item())
                cross_curvature_matrix[target_ord][source_ord] = cij
                if target_ord == source_ord:
                    target_info.update({
                        "curvature_method": "exact_hvp",
                        "hvp_frob_norm": float(torch.linalg.vector_norm(hv_target).item()),
                        "v_h_v": cij,
                        "v_h_v_raw_values": [cij],
                        "v_h_v_std": None,
                    })
            if torch.cuda.is_available():
                torch.cuda.empty_cache()

        for target_ord, (target_p, *_target_rest) in enumerate(block_params):
            target_info = block_info[id(target_p)]
            wi = target_info["w_norm"]
            mi = float(target_info["m_i"])
            kind_i = target_info["type"]
            shape_wi = wi / math.sqrt(max(mi, 1e-30))
            typed_cross_sum = 0.0
            all_cross_sum = 0.0
            for source_ord, (source_p, *_source_rest) in enumerate(block_params):
                if source_ord == target_ord:
                    continue
                source_info = block_info[id(source_p)]
                wj = source_info["w_norm"]
                mj = float(source_info["m_i"])
                cij = cross_curvature_matrix[target_ord][source_ord]
                if source_info["type"] == kind_i:
                    typed_den = wi + wj
                    if abs(typed_den) > eps_den:
                        typed_cross_sum += (2.0 * wj / typed_den) * cij
                shape_wj = wj / math.sqrt(max(mj, 1e-30))
                all_den = shape_wi + shape_wj
                if abs(all_den) > eps_den:
                    all_cross_sum += (
                        2.0
                        * math.sqrt(max(mj / max(mi, 1e-30), 0.0))
                        * shape_wj
                        / all_den
                    ) * cij
            self_curv = target_info["v_h_v"]
            target_info["cross_curvature_typed_sum"] = typed_cross_sum
            target_info["cross_curvature_all_sum"] = all_cross_sum
            target_info["weighted_adjusted_curvature_typed"] = self_curv + typed_cross_sum
            target_info["weighted_adjusted_curvature_all"] = self_curv + all_cross_sum

    elif curvature_method == "exact_hvp":
        for block_ord, (p, name, kind, block_idx, _lr, _wd) in enumerate(block_params):
            info = block_info[id(p)]
            v_i = info["v_i_tensor"]
            if dist.get_rank() == 0 and len(block_params) > 8 and block_ord % 12 == 0:
                print0(
                    f"local_quadratic_diag step:{step} hvp_block:{block_ord + 1}/{len(block_params)} {name}",
                    console=True,
                    log=False,
                )
            h_v = _exact_block_hessian_vec(
                model=model,
                diag_inputs=diag_inputs,
                diag_targets=diag_targets,
                param=p,
                vector=v_i,
                num_tokens_diag=num_tokens_diag,
                hvp_microbatch_size=hvp_microbatch_size,
            )
            v_h_v = float(torch.sum(v_i * h_v).item())
            info.update({
                "curvature_method": "exact_hvp",
                "hvp_frob_norm": float(torch.linalg.vector_norm(h_v).item()),
                "v_h_v": v_h_v,
                "v_h_v_raw_values": [v_h_v],
                "v_h_v_std": None,
            })
            if torch.cuda.is_available():
                torch.cuda.empty_cache()

    elif curvature_method == "block_loss_fd":
        for block_ord, (p, name, kind, block_idx, _lr, _wd) in enumerate(block_params):
            info = block_info[id(p)]
            v_i = info["v_i_tensor"].to(device=p.device, dtype=p.dtype)
            eps_i = float(info["fd_epsilon"])
            if dist.get_rank() == 0 and len(block_params) > 8 and block_ord % 12 == 0:
                print0(
                    f"local_quadratic_diag step:{step} loss_fd_block:{block_ord + 1}/{len(block_params)} {name}",
                    console=True,
                    log=False,
                )
            with torch.no_grad():
                p.add_(v_i, alpha=eps_i)
            f_plus_sum, f_plus = compute_diag_loss_only()
            with torch.no_grad():
                p.add_(v_i, alpha=-2.0 * eps_i)
            f_minus_sum, f_minus = compute_diag_loss_only()
            with torch.no_grad():
                p.add_(v_i, alpha=eps_i)

            v_h_v = (f_plus - 2.0 * f_0 + f_minus) / max(eps_i * eps_i, 1e-30)
            info.update({
                "curvature_method": "block_loss_fd",
                "hvp_frob_norm": None,
                "v_h_v": v_h_v,
                "v_h_v_raw_values": [v_h_v],
                "v_h_v_std": None,
                "fd_loss_base": f_0,
                "fd_loss_base_sum": f_0_sum,
                "fd_loss_plus": f_plus,
                "fd_loss_minus": f_minus,
                "fd_loss_plus_sum": f_plus_sum,
                "fd_loss_minus_sum": f_minus_sum,
            })

    elif curvature_method == "random_grad_fd":
        probe_count = max(1, int(fd_probes))
        curv_values: dict[int, list[float]] = {id(p): [] for p, *_ in block_params}
        gen = torch.Generator(device="cpu")
        gen.manual_seed(int(seed * 1_000_003 + step * 9_176 + diag_index * 101))

        def apply_probe(signs: list[float], scale: float) -> None:
            with torch.no_grad():
                for sign, (p, *_rest) in zip(signs, block_params):
                    info = block_info[id(p)]
                    eps_i = info["fd_epsilon"]
                    v_i = info["v_i_tensor"].to(device=p.device, dtype=p.dtype)
                    p.add_(v_i, alpha=float(scale) * float(sign) * float(eps_i))

        for probe_idx in range(probe_count):
            signs_tensor = torch.randint(
                0, 2, (len(block_params),), generator=gen, dtype=torch.int8
            )
            signs = [1.0 if int(v) else -1.0 for v in signs_tensor.tolist()]
            apply_probe(signs, +1.0)
            f_plus_sum, f_plus, g_plus = compute_diag_gradient()
            apply_probe(signs, -2.0)
            f_minus_sum, f_minus, g_minus = compute_diag_gradient()
            apply_probe(signs, +1.0)
            fd_probe_losses.append({
                "probe_index": int(probe_idx),
                "loss_plus": _safe_float(f_plus),
                "loss_minus": _safe_float(f_minus),
                "loss_plus_sum": _safe_float(f_plus_sum),
                "loss_minus_sum": _safe_float(f_minus_sum),
            })

            for sign, (p, *_rest) in zip(signs, block_params):
                info = block_info[id(p)]
                pid = id(p)
                eps_i = info["fd_epsilon"]
                v_i = info["v_i_tensor"]
                grad_diff = g_plus[pid].float() - g_minus[pid].float()
                dot_diff = float(torch.sum(v_i * grad_diff).item())
                h_hat = float(sign) * dot_diff / max(2.0 * eps_i, 1e-30)
                curv_values[pid].append(h_hat)

        for p, name, kind, block_idx, _lr, _wd in block_params:
            info = block_info[id(p)]
            vals = curv_values[id(p)]
            mean_vhv = sum(vals) / len(vals) if vals else None
            std_vhv = (
                math.sqrt(sum((v - mean_vhv) ** 2 for v in vals) / len(vals))
                if vals and mean_vhv is not None else None
            )
            info.update({
                "curvature_method": "random_grad_fd",
                "hvp_frob_norm": None,
                "v_h_v": mean_vhv,
                "v_h_v_raw_values": vals,
                "v_h_v_std": std_vhv,
            })

    else:
        raise ValueError(f"unknown local_quadratic curvature method {curvature_method!r}")

    if curvature_method == "exact_hvp":
        nonzero_hvp_blocks = sum(
            1
            for info in block_info.values()
            if (info.get("hvp_frob_norm") is not None and info["hvp_frob_norm"] > eps_den)
        )
        if block_info and nonzero_hvp_blocks == 0:
            raise RuntimeError(
                "exact_hvp produced zero HVP for every measured block; "
                "this indicates a broken second-order graph, often from a compiled model path."
            )

    for p, *_rest in block_params:
        info = block_info[id(p)]
        v_h_v = info["v_h_v"]
        denominator = info["w_norm"] * v_h_v if v_h_v is not None else None
        valid_t = (
            v_h_v is not None
            and math.isfinite(info["g_dot_v"])
            and math.isfinite(denominator)
            and abs(denominator) > eps_den
        )
        info["denominator_w_norm_v_h_v"] = denominator
        info["t_i"] = info["g_dot_v"] / denominator if valid_t else None
        info["t_i_valid"] = bool(valid_t)

    r_bar_C: dict[str, float] = {}
    for kind, pids in by_kind.items():
        r_bar_C[kind] = sum(block_info[pid]["r_i"] for pid in pids) / len(pids)

    r_bar_all = (
        sum(info["r_i"] for info in block_info.values()) / max(len(block_info), 1)
    )

    blocks_out = []
    t_vals = []
    x_plain = []
    x_typed = []
    x_all = []

    for p, name, kind, block_idx, _lr, _wd in block_params:
        info = block_info[id(p)]
        rc = r_bar_C[kind]
        ra = r_bar_all
        ti = info["t_i"]
        if ti is not None:
            t_vals.append(ti)
            x_plain.append(info["r_i"])
            x_typed.append(rc)
            x_all.append(ra)

        blocks_out.append({
            "block_name": name,
            "type": kind,
            "block_idx": block_idx,
            "m_i": info["m_i"],
            "n_i": info["n_i"],
            "d_i": info["d_i"],
            "w_norm": _safe_float(info["w_norm"]),
            "r_i": _safe_float(info["r_i"]),
            "rbar_C": _safe_float(rc),
            "rbar_all": _safe_float(ra),
            "x_plain": _safe_float(info["r_i"]),
            "x_typed": _safe_float(rc),
            "x_all": _safe_float(ra),
            "t_i": _safe_float_or_none(ti),
            "t_i_valid": bool(info["t_i_valid"]),
            "g_dot_v": _safe_float(info["g_dot_v"]),
            "v_h_v": _safe_float_or_none(info["v_h_v"]),
            "cross_curvature_typed_sum": _safe_float_or_none(info.get("cross_curvature_typed_sum")),
            "cross_curvature_all_sum": _safe_float_or_none(info.get("cross_curvature_all_sum")),
            "weighted_adjusted_curvature_typed": _safe_float_or_none(
                info.get("weighted_adjusted_curvature_typed")
            ),
            "weighted_adjusted_curvature_all": _safe_float_or_none(
                info.get("weighted_adjusted_curvature_all")
            ),
            "v_h_v_raw_values": [
                _safe_float(v) for v in info.get("v_h_v_raw_values", [])
                if math.isfinite(float(v))
            ],
            "v_h_v_std": _safe_float_or_none(info.get("v_h_v_std")),
            "denominator_w_norm_v_h_v": _safe_float_or_none(info["denominator_w_norm_v_h_v"]),
            "hvp_frob_norm": _safe_float_or_none(info["hvp_frob_norm"]),
            "curvature_method": info["curvature_method"],
            "fd_epsilon": _safe_float(info["fd_epsilon"]),
            "fd_loss_base": _safe_float_or_none(info.get("fd_loss_base")),

def _state_float_or_none(state: dict, key: str) -> float | None:
    value = state.get(key)

        return None

def _local_quadratic_grad_window_centers_for_step(step_index: int) -> list[int]:
    if not LOCAL_QUADRATIC_GRAD_WINDOW_ENABLED:
        return []
    radius = int(LOCAL_QUADRATIC_GRAD_WINDOW_RADIUS)
    lo = max(DIAG_FREQ, step_index - radius)
    hi = min(train_steps, step_index + radius)
    first = ((lo + DIAG_FREQ - 1) // DIAG_FREQ) * DIAG_FREQ
    centers = []
    center = first
    while center <= hi:
        if center != step_index and abs(step_index - center) <= radius:
            centers.append(int(center))
        center += DIAG_FREQ
    return centers


@torch.no_grad()
def _collect_grad_nuclear_window_step(step_index: int, centers: list[int]) -> dict:
    blocks = []
    for group in optimizer2.param_groups:
        for p in group["params"]:
            if p.ndim < 2 or p.grad is None:
                continue
            name = optimizer2._param_name_by_id.get(id(p), f"id_{id(p)}")
            kind = optimizer2._matrix_kind(p)
            block_idx = optimizer2._param_block_by_id.get(id(p), -1)
            g_mat = _tensor_to_matrix(p.grad.detach().float())
            nuc = float(torch.linalg.matrix_norm(g_mat, ord="nuc").item())
            blocks.append({
                "param_name": name,
                "type": kind,
                "block_idx": int(block_idx),
                "grad_nuclear_norm": _safe_float(nuc),
            })
    blocks.sort(key=lambda b: (b["type"], b["block_idx"], b["param_name"]))
    return {
        "step": int(step_index),
        "centers": [int(c) for c in centers],
        "blocks": blocks,
    }


def _compute_grad_window_criterion(diag_record: dict, window_records: list[dict]) -> dict:
    center_step = int(diag_record["step"])
    expected_steps = (
        list(range(center_step - LOCAL_QUADRATIC_GRAD_WINDOW_RADIUS, center_step))
        + list(range(center_step + 1, center_step + LOCAL_QUADRATIC_GRAD_WINDOW_RADIUS + 1))
    )
    records_by_step = {int(rec["step"]): rec for rec in window_records}
    actual_steps = sorted(records_by_step)
    missing_steps = [s for s in expected_steps if s not in records_by_step]

    blocks = [
        b for b in diag_record.get("blocks", [])
        if (
            b.get("v_h_v") is not None
            and b.get("w_norm") is not None
            and b.get("m_i") is not None
        )
    ]
    if LOCAL_QUADRATIC_GRAD_CRITERION_MODE == "all_layer_m":
        grouped_blocks: dict[str, list[dict]] = {"all_layers": blocks}
    else:
        grouped_blocks = {}
        for block in blocks:
            grouped_blocks.setdefault(block["type"], []).append(block)

    groups = {}
    for kind in sorted(grouped_blocks):
        layer_blocks = grouped_blocks[kind]
        block_names = [b["block_name"] for b in layer_blocks]
        w_by_name = {b["block_name"]: float(b["w_norm"]) for b in layer_blocks}
        l_by_name = {b["block_name"]: float(b["v_h_v"]) for b in layer_blocks}
        m_by_name = {b["block_name"]: float(b["m_i"]) for b in layer_blocks}
        sum_l = sum(l_by_name.values())
        sum_l_m = sum(l_by_name[name] * m_by_name[name] for name in block_names)
        sum_l_w2 = sum(l_by_name[name] * (w_by_name[name] ** 2) for name in block_names)
        lhs_den = sum_l_m if LOCAL_QUADRATIC_GRAD_CRITERION_MODE == "all_layer_m" else sum_l
        lhs = (
            sum_l_w2 / lhs_den
            if abs(lhs_den) > LOCAL_QUADRATIC_GRAD_NUC_EPS
            else float("nan")
        )

        numerator_sum = 0.0
        denominator_sum = 0.0
        grad_nuc_sum_by_name = {name: 0.0 for name in block_names}
        valid_window_steps = []
        for step_i in actual_steps:
            grad_blocks = {
                b["param_name"]: b
                for b in records_by_step[step_i].get("blocks", [])
                if b.get("grad_nuclear_norm") is not None
            }
            den_t = 0.0
            num_t = 0.0
            complete = True
            grad_nuc_this_step = {}
            for name in block_names:
                g_entry = grad_blocks.get(name)
                if g_entry is None:
                    complete = False
                    break
                g_nuc = float(g_entry["grad_nuclear_norm"])
                grad_nuc_this_step[name] = g_nuc
                den_t += g_nuc
                if LOCAL_QUADRATIC_GRAD_CRITERION_MODE == "all_layer_m":
                    num_t += (w_by_name[name] / math.sqrt(m_by_name[name])) * g_nuc
                else:
                    num_t += w_by_name[name] * g_nuc
            if complete and den_t > LOCAL_QUADRATIC_GRAD_NUC_EPS:
                valid_window_steps.append(int(step_i))
                numerator_sum += num_t
                denominator_sum += den_t
                for name, g_nuc in grad_nuc_this_step.items():
                    grad_nuc_sum_by_name[name] += g_nuc

        grad_weighted_mean = (
            numerator_sum / denominator_sum
            if denominator_sum > LOCAL_QUADRATIC_GRAD_NUC_EPS
            else float("nan")
        )
        rhs = grad_weighted_mean ** 2 if math.isfinite(grad_weighted_mean) else float("nan")
        ratio = (
            lhs / rhs
            if math.isfinite(lhs) and math.isfinite(rhs) and abs(rhs) > LOCAL_QUADRATIC_GRAD_NUC_EPS
            else float("nan")
        )
        criterion_valid = bool(
            math.isfinite(lhs)
            and math.isfinite(rhs)
            and abs(rhs) > LOCAL_QUADRATIC_GRAD_NUC_EPS
        )
        zero_over_zero = bool(
            math.isfinite(lhs)
            and math.isfinite(rhs)
            and abs(lhs) <= LOCAL_QUADRATIC_GRAD_NUC_EPS
            and abs(rhs) <= LOCAL_QUADRATIC_GRAD_NUC_EPS
        )
        problem_layers = []
        for name in block_names:
            issues = []
            w_norm = w_by_name[name]
            l_i = l_by_name[name]
            grad_nuc_sum = grad_nuc_sum_by_name[name]
            if abs(w_norm) <= LOCAL_QUADRATIC_GRAD_NUC_EPS:
                if abs(l_i) > LOCAL_QUADRATIC_GRAD_NUC_EPS:
                    issues.append("zero_weight_nonzero_curvature")
                if grad_nuc_sum > LOCAL_QUADRATIC_GRAD_NUC_EPS:
                    issues.append("zero_weight_nonzero_grad_window")
            if issues:
                problem_layers.append({
                    "block_name": name,
                    "issues": issues,
                    "w_norm": _safe_float(w_norm),
                    "m_i": _safe_float(m_by_name[name]),
                    "L_i_v_h_v": _safe_float(l_i),
                    "grad_nuclear_norm_window_sum": _safe_float(grad_nuc_sum),
                })
        problem_layers.sort(
            key=lambda layer: (
                -float(layer["grad_nuclear_norm_window_sum"] or 0.0),
                layer["block_name"],
            )
        )
        warning_kind = None
        warning_message = None
        if zero_over_zero:
            warning_kind = "criterion_zero_over_zero"
            warning_message = (
                "Criterion is 0/0: both lhs and rhs are numerically zero. "
                "Inspect problem_layers for zero-norm weights carrying curvature or gradient signal."
            )
        elif not criterion_valid:
            warning_kind = "criterion_invalid"
            warning_message = (
                "Criterion ratio is invalid because lhs or rhs is non-finite, "
                "or rhs is numerically zero."
            )
        elif problem_layers:
            warning_kind = "problem_layers_present"
            warning_message = (
                "Some layers have zero-norm weights with nonzero curvature or "
                "gradient-window signal; criterion is finite but should be inspected."
            )
        is_all_layer_m = LOCAL_QUADRATIC_GRAD_CRITERION_MODE == "all_layer_m"
        groups[kind] = {
            "criterion_mode": LOCAL_QUADRATIC_GRAD_CRITERION_MODE,
            "criterion_valid": criterion_valid,
            "zero_over_zero": zero_over_zero,
            "warning_kind": warning_kind,
            "warning_message": warning_message,
            "warning_problem_layer_count": len(problem_layers),
            "warning_problem_layers": problem_layers,
            "n_layers": len(layer_blocks),
            "sum_L": _safe_float(sum_l),
            "sum_L_m_i": _safe_float(sum_l_m),
            "sum_L_w_norm_sq": _safe_float(sum_l_w2),
            "rhs_grad_nuclear_norm_sum": _safe_float(denominator_sum),
            "rhs_weighted_grad_nuclear_norm_sum": _safe_float(numerator_sum),
            "lhs": _safe_float(lhs),
            "lhs_denominator": "sum_i L_i m_i" if is_all_layer_m else "sum_i L_i",
            "lhs_sum_L_w2_over_sum_L": _safe_float(lhs) if not is_all_layer_m else None,
            "lhs_sum_L_w2_over_sum_L_m": _safe_float(lhs) if is_all_layer_m else None,
            "grad_window_step_count": len(valid_window_steps),
            "grad_window_steps": valid_window_steps,
            "rhs": _safe_float(rhs),
            "rhs_numerator": (
                "E_t sum_i (||W_i||_F / sqrt(m_i)) ||G_i,t||_*"
                if is_all_layer_m
                else "E_t sum_i ||W_i||_F ||G_i,t||_*"
            ),
            "rhs_denominator": "E_t sum_i ||G_i,t||_*",
            "rhs_grad_weighted_w_mean_sq": _safe_float(rhs) if not is_all_layer_m else None,
            "rhs_grad_weighted_w_over_sqrt_m_mean_sq": _safe_float(rhs) if is_all_layer_m else None,
            "lhs_over_rhs": _safe_float(ratio),
            "criterion_lhs_le_rhs": bool(criterion_valid and lhs <= rhs),
            "layers": [
                {
                    "block_name": name,
                    "w_norm": _safe_float(w_by_name[name]),
                    "m_i": _safe_float(m_by_name[name]),
                    "L_i_v_h_v": _safe_float(l_by_name[name]),
                    "grad_nuclear_norm_window_sum": _safe_float(grad_nuc_sum_by_name[name]),
                }
                for name in block_names
            ],
        }

    return {
        "center_step": center_step,
        "criterion_mode": LOCAL_QUADRATIC_GRAD_CRITERION_MODE,
        "radius": int(LOCAL_QUADRATIC_GRAD_WINDOW_RADIUS),
        "expected_window_step_count": len(expected_steps),
        "actual_window_step_count": len(actual_steps),
        "missing_window_steps": missing_steps,
        "groups": groups,
    }


def _write_grad_window_outputs(
    diag_records: list[dict],
    grad_window_by_center: dict[int, list[dict]],
    criteria_by_center: dict[int, dict],
) -> None:
    if not (dist.get_rank() == 0 and MYOPT_SAVE_DIAG):
        return
    if diag_file is not None:
        _write_json_atomic(diag_file, diag_records)
    if grad_window_file is not None:
        raw_records = []
        for center in sorted(grad_window_by_center):
            for record in sorted(grad_window_by_center[center], key=lambda r: int(r["step"])):
                out = dict(record)
                out["center_step"] = int(center)
                out["offset_from_center"] = int(record["step"]) - int(center)
                raw_records.append(out)
        _write_json_atomic(grad_window_file, raw_records)
    if grad_window_criteria_file is not None:
        _write_json_atomic(
            grad_window_criteria_file,
            [criteria_by_center[k] for k in sorted(criteria_by_center)],
        )
    if grad_window_criteria_csv is not None:
        with grad_window_criteria_csv.open("w", newline="") as f:
            fieldnames = [
                "center_step",
                "type",
                "criterion_mode",
                "n_layers",
                "sum_L",
                "sum_L_m_i",
                "sum_L_w_norm_sq",
                "rhs_grad_nuclear_norm_sum",
                "rhs_weighted_grad_nuclear_norm_sum",
                "lhs",
                "lhs_sum_L_w2_over_sum_L",
                "lhs_sum_L_w2_over_sum_L_m",
                "rhs",
                "rhs_grad_weighted_w_mean_sq",
                "rhs_grad_weighted_w_over_sqrt_m_mean_sq",
                "lhs_over_rhs",
                "criterion_valid",
                "zero_over_zero",
                "criterion_lhs_le_rhs",
                "warning_kind",
                "warning_message",
                "warning_problem_layer_count",
                "warning_problem_layers",
                "grad_window_step_count",
                "expected_window_step_count",
                "actual_window_step_count",
            ]
            writer = csv.DictWriter(f, fieldnames=fieldnames)
            writer.writeheader()
            for center in sorted(criteria_by_center):
                crit = criteria_by_center[center]
                for kind, group in sorted(crit["groups"].items()):
                    writer.writerow({
                        "center_step": center,
                        "type": kind,
                        "criterion_mode": group["criterion_mode"],

def schedule_multiplier(step, cooldown_frac=None):
    cooldown_frac = MYOPT_COOLDOWN_FRAC if cooldown_frac is None else float(cooldown_frac)
    progress = step / schedule_steps
    return 1.0 if progress < 1 - cooldown_frac else (1 - progress) / cooldown_frac

def set_hparams(step, cooldown_frac=None):
    eta = schedule_multiplier(step, cooldown_frac)
    for opt in optimizers:
        for group in opt.param_groups:
            group["lr"] = group["initial_lr"] * eta

def finalize_ready_grad_windows(current_step: int) -> None:
    if not (dist.get_rank() == 0 and MYOPT_SAVE_DIAG and LOCAL_QUADRATIC_GRAD_WINDOW_ENABLED):
        return
    changed = False
    for center in sorted(grad_window_by_center):
        if center in grad_window_criteria_by_center:
            continue
        if current_step < center + LOCAL_QUADRATIC_GRAD_WINDOW_RADIUS:
            continue
        diag_record = diag_record_by_step.get(center)
        if diag_record is None:
            continue
        criterion = _compute_grad_window_criterion(diag_record, grad_window_by_center[center])
        diag_record["grad_window_criterion"] = criterion
        grad_window_criteria_by_center[center] = criterion
        for kind, group in sorted(criterion.get("groups", {}).items()):
            if group.get("warning_kind"):
                problem_layers = group.get("warning_problem_layers", [])
                shown_layers = [layer["block_name"] for layer in problem_layers[:12]]
                suffix = "" if len(problem_layers) <= 12 else f"; +{len(problem_layers) - 12} more"
                print0(
                    "WARNING grad_window_criterion "
                    f"step:{center} group:{kind} mode:{criterion.get('criterion_mode')} "
                    f"warning:{group.get('warning_kind')} "
                    f"valid:{group.get('criterion_valid')} "
                    f"zero_over_zero:{group.get('zero_over_zero')} "
                    f"lhs:{group.get('lhs')} rhs:{group.get('rhs')} "
                    f"problem_layers:{', '.join(shown_layers)}{suffix}",
                    console=True,
                )
        changed = True
    if changed:
        _write_grad_window_outputs(
            diag_records,
            grad_window_by_center,
            grad_window_criteria_by_center,
        )

for step in range(train_steps + 1):

    # -------- validation --------
    val_step_freq = MID_VAL_STEP_FREQ if step >= MID_VAL_START_STEP else EARLY_VAL_STEP_FREQ
    if step == train_steps or step % val_step_freq == 0:
        dist.barrier()
        time_since_last_val = time.perf_counter() - t0
        step_avg = time_since_last_val / (step - last_val_step) if step > 0 else float("nan")
        last_val_step = step
        training_time += time_since_last_val
        model.eval()
        train_model.eval()
        val_loss = torch.zeros((), device=device)
        with torch.no_grad():
            for i in range(len(val_inputs) // train_microbatch_sequences):
                start = i * train_microbatch_sequences
                end = (i + 1) * train_microbatch_sequences
                val_loss += train_model(val_inputs[start:end], val_targets[start:end])
        dist.all_reduce(val_loss, op=dist.ReduceOp.SUM)
        val_loss_per_tok = float(val_loss) / val_tokens
        print0(f"step:{step}/{train_steps} val_loss:{val_loss_per_tok:.5f}"
               f" train_time:{training_time:.3f}s step_avg:{1000*step_avg:.2f}ms", console=True)
        model.train()
        train_model.train()
        dist.barrier()
        t0 = time.perf_counter()

    if step == train_steps:
        break

    # -------- training step --------
    inputs, targets = next(train_loader)
    assert len(inputs) % train_microbatches == 0
    current_microbatch_sequences = len(inputs) // train_microbatches
    assert current_microbatch_sequences == train_microbatch_sequences
    step_train_loss = 0.0
    for i in range(train_microbatches):
        start = i * current_microbatch_sequences
        end = (i + 1) * current_microbatch_sequences
        loss = train_model(inputs[start:end], targets[start:end])
        step_train_loss += loss.item()
        loss.backward()
    step_train_loss /= (len(inputs) * sequence_length)  # per token

    for name_p, p in model.named_parameters():
        assert p.grad is not None, name_p
        dist.all_reduce(p.grad, op=dist.ReduceOp.SUM)
    current_step_index = step + 1
    if LOCAL_QUADRATIC_GRAD_WINDOW_ENABLED:

        if centers and dist.get_rank() == 0:
            grad_window_record = _collect_grad_nuclear_window_step(current_step_index, centers)
            for center in centers:
                grad_window_by_center.setdefault(center, []).append(grad_window_record)
    set_hparams(step)
    current_lr = optimizer2.param_groups[0]["lr"]

    if MYOPT_SAVE_DIAG and (step + 1) % DIAG_FREQ == 0:

                l_refresh_diag_interval=ORSCALE_L_REFRESH_DIAG_INTERVAL,
